import test from 'node:test';
import assert from 'node:assert/strict';
import { mkdtempSync, readFileSync, statSync, mkdirSync, writeFileSync } from 'node:fs';
import { tmpdir } from 'node:os';
import path from 'node:path';
import { casesAllowed, runAnalysisProbe } from './report-analysis-spike.mjs';

const secret = 'synthetic-test-value-never-sent-to-any-provider';
function harness(runMaven = () => {}) {
  const root = mkdtempSync(path.join(tmpdir(), 'majorpilot-analysis-guard-test-'));
  const calls = { env: 0, runtime: 0, maven: 0, log: [], errors: [] };
  const dependencies = {
    root,
    readEnv: () => { calls.env++; return { BAILIAN_API_KEY: secret, DEEPSEEK_API_KEY: secret, MP_REPORT_AI_MODEL: 'synthetic-model' }; },
    buildRuntimeEnv: () => { calls.runtime++; return { PATH: '/synthetic/no-executables' }; },
    runMaven: (...args) => { calls.maven++; runMaven(...args); },
    log: value => calls.log.push(value), error: value => calls.errors.push(value),
  };
  return { root, calls, dependencies };
}
const command = (cases = 'ba-good', id = 'offline-test') => ['--live', '--cases', cases, '--run-id', id];

test('--check has no environment, filesystem guard, Maven or paid-call side effects', () => {
  const h = harness(() => assert.fail('No subprocess allowed'));
  const output = runAnalysisProbe(['--check'], h.dependencies);
  assert.equal(output.exitCode, 0); assert.equal(output.result.paidCalls, 0);
  assert.equal(h.calls.env, 0); assert.equal(h.calls.runtime, 0); assert.equal(h.calls.maven, 0);
  assert.deepEqual(output.result.casesAllowed, casesAllowed);
  assert.equal(output.result.maxCases, 6); assert.equal(output.result.maxHttpAttemptsPerCase, 3);
  assert.equal(output.result.mandatoryIndependentRewrite, true);
  assert.equal(output.result.maxDurationMsPerCase, 2_700_000);
});

test('invalid or incomplete arguments fail before secrets or executable access', () => {
  for (const args of [[], ['--live'], ['--check', '--live'], command('unknown'), command('ba-good', '../escape'),
    ['--live', '--cases', 'ba-good', '--cases', 'ba-good'], ['--live', '--cases', 'ba-good', '--unsafe', 'x'],
    [...command(), '--provider', 'OTHER'], ['--live', '--provider', 'BAILIAN', '--run-id', 'offline-test']]) {
    const h = harness(() => assert.fail('No subprocess allowed'));
    assert.throws(() => runAnalysisProbe(args, h.dependencies));
    assert.equal(h.calls.env, 0, JSON.stringify(args)); assert.equal(h.calls.maven, 0);
  }
});

test('repeating a case or inserting an empty case is forbidden', () => {
  for (const cases of ['ba-good,ba-good', 'ba-good,', ',ba-good', casesAllowed.join(',') + ',ba-good']) {
    const h = harness(() => assert.fail('No subprocess allowed'));
    assert.throws(() => runAnalysisProbe(command(cases), h.dependencies));
    assert.equal(h.calls.env, 0); assert.equal(h.calls.maven, 0);
  }
});

test('explicit allowed cases dispatch one isolated Maven invocation and private bounded guard', () => {
  let observed;
  const h = harness((executable, args, options) => { observed = { executable, args, options }; });
  const result = runAnalysisProbe(command('ba-good,ai-metric-error'), h.dependencies);
  assert.equal(result.exitCode, 0); assert.equal(h.calls.maven, 1);
  assert.equal(observed.executable, path.join(h.root, 'backend/mvnw'));
  assert.ok(observed.args.includes('-Dtest=ReportAnalysisLiveTest'));
  assert.equal(observed.options.env.MP_REPORT_ANALYSIS_LIVE_CASES, 'ba-good,ai-metric-error');
  assert.equal(observed.options.env.MP_REPORT_ANALYSIS_LIVE_CONFIRM, 'synthetic-quality-only');
  for (const key of ['MP_RUN_MYSQL_TESTS', 'MP_RUN_LIVE_REPORT_TESTS', 'MP_RUN_BAILIAN_TEST']) assert.equal(observed.options.env[key], 'false');
  assert.equal(observed.options.timeout, 2 * 2_700_000 + 180000);
  assert.equal(result.metadata.maxHttpAttempts, 6);
  assert.equal(result.metadata.mandatoryIndependentRewrite, true);
  assert.equal(result.metadata.maxDurationMsPerCase, 2_700_000);
  assert.equal(result.metadata.maxTotalDurationMs, 2 * 2_700_000);
  assert.equal(statSync(result.guard).mode & 0o777, 0o600);
  assert.equal(statSync(path.dirname(result.guard)).mode & 0o777, 0o700);
  assert.equal(result.metadata.status, 'REWRITTEN_VALIDATED_REQUIRES_HUMAN_SEMANTIC_REVIEW');
  assert.equal(readFileSync(result.guard, 'utf8').includes(secret), false);
});

test('existing guard content cannot be replaced and prevents another invocation', () => {
  const h = harness();
  const first = runAnalysisProbe(command(), h.dependencies);
  const before = readFileSync(first.guard, 'utf8');
  assert.throws(() => runAnalysisProbe(command(), h.dependencies), { code: 'EEXIST' });
  assert.equal(h.calls.maven, 1); assert.equal(readFileSync(first.guard, 'utf8'), before);
});

test('failure or timeout consumes the run ID with no automatic resend or error-content leak', () => {
  const h = harness(() => { throw new Error(`Simulated timeout ${secret}`); });
  const result = runAnalysisProbe(command(), h.dependencies);
  assert.equal(result.exitCode, 1); assert.equal(h.calls.maven, 1);
  assert.equal(result.metadata.status, 'FAILED_OR_TIMED_OUT_DO_NOT_AUTOMATICALLY_REPEAT');
  const before = readFileSync(result.guard, 'utf8');
  assert.throws(() => runAnalysisProbe(command(), h.dependencies), { code: 'EEXIST' });
  assert.equal(h.calls.maven, 1); assert.equal(readFileSync(result.guard, 'utf8'), before);
  assert.equal(h.calls.errors.join('').includes(secret), false);
});

test('unprotected guard directory and absent provider key fail closed', () => {
  const h = harness();
  const directory = path.join(h.root, 'data/report-analysis-probes');
  mkdirSync(directory, { recursive: true, mode: 0o755 });
  assert.throws(() => runAnalysisProbe(command(), h.dependencies)); assert.equal(h.calls.maven, 0);
  const missing = harness(); missing.dependencies.readEnv = () => ({});
  assert.throws(() => runAnalysisProbe(command(), missing.dependencies), /key is missing/); assert.equal(missing.calls.maven, 0);
});

test('preexisting reserved guard is retained even if no prior process reached Maven', () => {
  const h = harness();const directory = path.join(h.root, 'data/report-analysis-probes');mkdirSync(directory, { recursive: true, mode: 0o700 });
  const guard = path.join(directory, 'bailian-offline-test.attempt.json');writeFileSync(guard, '{"status":"ATTEMPT_RESERVED"}', { mode: 0o600 });
  assert.throws(() => runAnalysisProbe(command(), h.dependencies), { code: 'EEXIST' });assert.equal(h.calls.maven, 0);
  assert.equal(readFileSync(guard, 'utf8'), '{"status":"ATTEMPT_RESERVED"}');
});
