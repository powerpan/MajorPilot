function sleep(ms) { return new Promise(resolve => setTimeout(resolve, ms)); }
function openTextFields(value, schema, path = '') {
  if (Array.isArray(value)) return value.flatMap((item, index) => openTextFields(item, schema.items || {}, `${path}/${index}`));
  if (value && typeof value === 'object') return Object.entries(value).flatMap(([field, item]) =>
    schema.properties?.[field] ? openTextFields(item, schema.properties[field], path ? `${path}/${field}` : field) : []);
  return typeof value === 'string' && value.trim() && !schema.enum && !Object.hasOwn(schema, 'const') ? [{ field: path, value }] : [];
}
function assertAnalysis(body) {
  if (body.schemaVersion === 'report-v1') {
    assert.equal(Object.hasOwn(body, 'analysis'), false, 'Historical v1 must not acquire invented analysis');
    return;
  }
  assert.equal(body.schemaVersion, 'report-v2');
  assert.ok(body.analysis && typeof body.analysis.summary === 'string');
  const analysis = body.analysis, complete = body.generationMode === 'AI';
  assert.equal(analysis.status, complete ? 'COMPLETE' : 'UNAVAILABLE');
  if (!complete) {
    for (const key of ['majorAnalyses', 'findings', 'crossMajorAnalysis']) assert.deepEqual(analysis[key], [], 'Fallback must not publish fabricated critique');
    return;
  }
  const majors = body.sources.map(source => source.majorCode).sort();
  assert.deepEqual(analysis.majorAnalyses.map(major => major.majorCode).sort(), majors, 'Every source Major needs its own analysis');
  for (const major of analysis.majorAnalyses) assert.equal(Object.hasOwn(major, 'evidenceRefs'), false, 'Major analysis must not expose internal source references');
  const required = new Set(), seen = new Set();
  for (const code of majors) {
    const material = materials.find(item => item.major === code); assert.ok(material);
    for (const step of material.manifest.steps) if (openTextFields(material.fixture.responses[step.stepKey], step.inputSchema).length) required.add(`${code}:${step.stepKey}`);
  }
  for (const finding of analysis.findings) {
    assert.ok(majors.includes(finding.majorCode), 'Finding must belong to a source Major');
    const material = materials.find(item => item.major === finding.majorCode);
    const definition = material.manifest.steps.find(step => step.stepKey === finding.stepKey); assert.ok(definition);
    const answer = openTextFields(material.fixture.responses[finding.stepKey], definition.inputSchema).find(item => item.field === finding.answerField);
    assert.ok(answer && typeof finding.answerQuote === 'string' && finding.answerQuote.length > 0 && answer.value.includes(finding.answerQuote), 'Finding must quote the exact synthetic answer in its own Major/step/field');
    const identity = `${finding.majorCode}:${finding.stepKey}:${finding.answerField}`;
    assert.ok(!seen.has(identity), 'Finding must not repeat the same answer field'); seen.add(identity);
    required.delete(`${finding.majorCode}:${finding.stepKey}`);
    assert.equal(Object.hasOwn(finding, 'evidenceRefs'), false, 'Internal references must not leak to the Student');
    for (const field of ['reasoning', 'improvement', 'example']) assert.ok(typeof finding[field] === 'string' && finding[field].trim());
  }
  assert.equal(required.size, 0, 'Every answered step must be analyzed, not merely the final reflection');
  if (body.reportType === 'EXPERIENCE') assert.deepEqual(analysis.crossMajorAnalysis, []);
  else {
    assert.ok(analysis.crossMajorAnalysis.length > 0, 'Comparison needs evidence-bound cross-Major interpretation');
    for (const comparison of analysis.crossMajorAnalysis) {
      assert.ok(comparison.majorCodes.length >= 2 && comparison.majorCodes.every(code => majors.includes(code)));
      assert.equal(Object.hasOwn(comparison, 'evidenceRefs'), false);
    }
  }
}
function gateway(mode) {
  assert.ok(['PASS', 'INVALID_INITIAL', 'INVALID_ALWAYS', 'INVALID_EXPERIENCE_ALWAYS', 'HOLD_INITIAL'].includes(mode));
  const base = `${ROOT}/data/p6-harness`, file = `${base}/gateway-control.json`, temporary = `${base}/control-${randomUUID()}.json`;
  for (const artifact of ['gateway-entered.json', 'gateway-release']) if (existsSync(`${base}/${artifact}`)) unlinkSync(`${base}/${artifact}`);
  writeFileSync(temporary, JSON.stringify({ mode }) + '\n', { mode: 0o600, flag: 'wx' }); renameSync(temporary, file);
}
function releaseGateway() { const file = `${ROOT}/data/p6-harness/gateway-release`; if (!existsSync(file)) writeFileSync(file, 'release\n', { flag: 'wx', mode: 0o600 }); }
async function until(check, label, timeout = 55000) {
  const deadline = Date.now() + timeout;
  do { const result = await check(); if (result) return result; await sleep(2000); } while (Date.now() < deadline);
  throw new Error(`${label}: timed out; no fixture payload emitted`);
}

    for (const [versionId, schema] of versions) {
      assert.match(versionId, /^[0-7][0-9A-HJKMNP-TV-Z]{25}$/);
      const bytes = p6Query(`SELECT HEX(input_manifest_json) FROM mp_report_version WHERE public_id='${versionId}' AND report_id IN (${ownerReports});`);
      assert.match(bytes, /^(?:[a-f0-9]{2})+$/i);
      const pinned = JSON.parse(Buffer.from(bytes, 'hex').toString('utf8'));
      if (schema === 'report-v1') {
        assert.ok(!JSON.stringify(pinned).includes('STUDENT_ANALYSIS'), 'v1 keeps its immutable historical projection');
        assert.ok(pinned.sources.every(source => !Object.hasOwn(source, 'analysisContext')));
        continue;
      }
      assert.equal(schema, 'report-v2'); assert.equal(pinned.versions.inputSchemaVersion, 'report-input-v2');
      for (const source of pinned.sources) {
        const material = materials.find(item => item.major === source.majorCode); assert.ok(material);
        const context = source.analysisContext;
        assert.equal(context.schemaVersion, 'report-analysis-context-v2');
        assert.ok(source.items.some(item => item.type === 'STUDENT_ANALYSIS'), 'v2 must include original unverified reasoning');
        for (const evidence of source.items.filter(item => item.type === 'STUDENT_ANALYSIS')) assert.deepEqual(evidence.allowedClaimTypes, [], 'Raw answers must not become verified facts');
        assert.equal(context.steps.length, material.manifest.steps.length);
        for (const step of context.steps) {
          const definition = material.manifest.steps.find(item => item.stepKey === step.stepKey); assert.ok(definition);
          assert.deepEqual(step.submittedResponse, material.fixture.responses[step.stepKey], 'Frozen input must retain the original answer, including numbers and choices');
          const expected = openTextFields(material.fixture.responses[step.stepKey], definition.inputSchema).sort((a, b) => a.field.localeCompare(b.field));
          const actual = step.answers.map(answer => ({ field: answer.answerField, value: answer.value })).sort((a, b) => a.field.localeCompare(b.field));
          assert.deepEqual(actual, expected, 'All relevant open fields must reach the analysis input without omission');
          assert.ok(step.answers.every(answer => Boolean(answer.evidenceId) !== Boolean(answer.sourceRef)), 'Every raw answer has exactly one authenticated reference');
          assert.ok(step.rubric.criteria.length > 0 && step.instructions, 'Task requirements and assessment criteria must accompany the answers');
        }
        assert.deepEqual(context.reflection.answers, material.fixture.reflection.answers);
        assert.ok(context.verifiedExecutions.length > 0, 'Model analysis includes actual server-verified execution context');
      }
    }
    const analysisLinks = `FROM mp_report_claim c JOIN mp_report_claim_evidence x ON x.report_claim_id=c.id JOIN mp_evidence_item e ON e.id=x.evidence_item_id JOIN mp_report_version v ON v.id=c.report_version_id WHERE v.report_id IN (${ownerReports}) AND e.evidence_type='STUDENT_ANALYSIS'`;
    assert.equal(scalar(`SELECT COUNT(*) ${analysisLinks} AND v.schema_version='report-v1';`), 0, 'Historical v1 claims never adopt raw analysis');
    assert.ok(scalar(`SELECT COUNT(*) ${analysisLinks} AND v.schema_version='report-v2';`) > 0, 'New analysis must actually cite raw responses');
    assert.equal(scalar(`SELECT COUNT(*) ${analysisLinks} AND NOT (v.schema_version='report-v2' AND v.generation_mode='AI' AND c.section_key='DEEP_ANALYSIS' AND c.claim_type='BOUNDED_INTERPRETATION' AND c.claim_rule_version='report-claims-v2');`), 0, 'Raw answers may only support versioned bounded critique, never verified fact claims');
    checks.push('feedback ownership/replay, observed page attribution, v1 projection preserved, complete v2 original-answer and rubric input, exact per-step quotation, bounded analysis citations, no paid supplier calls');
    const missing = contract.operations.filter(op => !verified.has(op.id)); assert.deepEqual(missing.map(op => op.id), [], 'Every declared P6 operation must be exercised');
  } catch (error) { failure = error; }
  finally {
    try {
      releaseGateway();
      // Leave release present while an in-flight fake may be unwinding. Account
      // locks in cleanup fence publication; only then reset the shared mode.
      for (const admin of adminClients) await admin.call('/logout', { method: 'POST' });
      await scope.cleanup(); await sleep(250); gateway('PASS');
    }
    catch (cleanup) { failure = failure ? new AggregateError([failure, cleanup], 'P6 acceptance/cleanup failed; payloads suppressed') : cleanup; }
  }
  if (failure) throw failure;
  console.log(JSON.stringify({ result: 'PASS', scope: 'P6 real isolated HTTP/MySQL with test-source-only published preconditions; not human review', checks, responseContractsVerified: verified.size, operationStatuses: Object.fromEntries([...verified].map(([id, statuses]) => [id, [...statuses].sort()])), fixtureAccounts: 2, fixtureCleanup: true, rateLimitsReset: false, simulatedGatewayOnly: true, paidCalls: 0 }, null, 2));
