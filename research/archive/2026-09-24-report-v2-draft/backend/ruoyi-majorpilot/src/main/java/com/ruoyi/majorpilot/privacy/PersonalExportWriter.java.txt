/** In-memory, explicit personal-data projection. The caller owns the Student lock and snapshot. */
@Service @Profile("local")
public class PersonalExportWriter {
    static final long MAX_BYTES=64L*1024*1024;
    private final JdbcTemplate db;private final ObjectMapper mapper;private final ReportSchema reportSchema;
    public PersonalExportWriter(JdbcTemplate db,ObjectMapper mapper){this.db=db;this.mapper=mapper.copy().disable(SerializationFeature.WRITE_DATES_AS_TIMESTAMPS);reportSchema=new ReportSchema(this.mapper);}
    public byte[] build(long studentId,String scope,Instant generatedAt){
        if(!TransactionSynchronizationManager.isActualTransactionActive())throw new IllegalStateException("PERSONAL_EXPORT_SNAPSHOT_REQUIRED");
        if(!Set.of("REPORTS_ONLY","FULL_PERSONAL_EXPORT").contains(scope==null?"":scope)||generatedAt==null)throw new ApiException(422,"VALIDATION_ERROR");
        var accounts=db.queryForList("SELECT public_id,email_display,status,email_verified_at,last_login_at,created_at,updated_at FROM mp_student_account WHERE id=?",studentId);
        if(accounts.size()!=1||!Set.of("ACTIVE","PENDING_VERIFICATION").contains(accounts.getFirst().get("status")))throw new ApiException(409,"ACCOUNT_UNAVAILABLE");
        var account=accounts.getFirst();var archive=new Archive(MAX_BYTES);var files=mapper.createArrayNode();var published=publishedReports(studentId);
        for(var report:published){String path="reports/"+report.versionId+".html";archive.add(path,html(report.body).getBytes(StandardCharsets.UTF_8));var row=mapper.createObjectNode();row.put("path",path).put("reportId",report.reportId).put("reportVersionId",report.versionId).put("versionNo",report.number).put("publicationStatus",report.status).put("reportType",report.body.path("reportType").asText());files.add(row);}
        if(scope.equals("FULL_PERSONAL_EXPORT")){var data=personal(studentId,account,generatedAt);archive.add("personal-data.json",bytes(data));files.addObject().put("path","personal-data.json").put("schemaVersion","personal-data-v1");}
        var manifest=mapper.createObjectNode();manifest.put("schemaVersion","personal-export-v1").put("scope",scope).put("generatedAt",generatedAt.toString()).put("studentId",account.get("public_id").toString());manifest.set("files",files);
        manifest.set("notes",mapper.valueToTree(List.of("每份报告保留其发布时的版本；旧版不代表当前最新版。","自己的自由文本按原文导出，不视为经过内容验证的事实。","DEMO 不表示专业完成；REVIEW_PREVIEW 保留原始模式，只有附有明确产品接入授权的既有记录才纳入报告流程。","本包不包含密码、会话、令牌、安全风控、管理审计、原始埋点、模型提示词、私有评价标签或他人的资料。")));
        archive.add("manifest.json",bytes(manifest));return archive.finish();
    }
    private ObjectNode personal(long student,Map<String,Object> account,Instant at){
        var out=mapper.createObjectNode();out.put("schemaVersion","personal-data-v1").put("generatedAt",at.toString());out.set("account",project(account,"public_id:studentId","email_display:email","status:status","email_verified_at:emailVerifiedAt","last_login_at:lastLoginAt","created_at:createdAt","updated_at:updatedAt"));
        out.set("consents",rows("SELECT c.document_type,c.document_version,c.decision,c.source,c.decided_at FROM mp_consent_record c LEFT JOIN mp_intake_session i ON i.id=c.intake_session_id WHERE c.student_id=? OR (c.student_id IS NULL AND i.claimed_student_id=?) ORDER BY c.id",new Object[]{student,student},"document_type:documentType","document_version:documentVersion","decision:decision","source:source","decided_at:decidedAt"));
        var intakes=mapper.createArrayNode();for(var r:db.queryForList("SELECT public_id,questionnaire_version,status,draft_json,created_at,claimed_at,last_step_key FROM mp_intake_session WHERE claimed_student_id=? ORDER BY id",student)){var item=project(r,"public_id:intakeId","questionnaire_version:questionnaireVersion","status:status","created_at:createdAt","claimed_at:claimedAt","last_step_key:lastStepKey");item.set("answers",questionnaireAnswers(node(r.get("draft_json"))));intakes.add(item);}out.set("claimedIntakes",intakes);
        var profiles=mapper.createArrayNode();for(var r:db.queryForList("SELECT p.id,p.public_id,p.profile_version_no,p.questionnaire_version,p.status,p.submitted_at,old.public_id AS supersedes_id FROM mp_decision_profile p LEFT JOIN mp_decision_profile old ON old.id=p.supersedes_profile_id AND old.student_id=p.student_id WHERE p.student_id=? ORDER BY p.profile_version_no,p.id",student)){var item=project(r,"public_id:decisionProfileId","profile_version_no:versionNo","questionnaire_version:questionnaireVersion","status:status","submitted_at:submittedAt","supersedes_id:supersedesProfileId");var answers=mapper.createArrayNode();for(var a:db.queryForList("SELECT question_key,answer_json,schema_version,created_at FROM mp_decision_answer WHERE decision_profile_id=? ORDER BY id",n(r,"id"))){String key=s(a,"question_key");if(!questionKeys().contains(key))throw invalid();var answer=project(a,"question_key:questionKey","schema_version:schemaVersion","created_at:answeredAt");answer.set("answer",node(a.get("answer_json")));answers.add(answer);}item.set("answers",answers);profiles.add(item);}out.set("profiles",profiles);out.set("questionnaire",mapper.valueToTree(QuestionnaireDefinition.current()));
        var recommendations=mapper.createArrayNode();for(var r:db.queryForList("SELECT r.id,r.public_id,p.public_id AS profile_public_id,x.public_id AS run_public_id,r.recommendation_type,r.completed_major_codes_json,r.rule_version,r.primary_major_code,r.alternative_order_json,r.tie_detected,r.tie_break_rule,r.explanation_status,r.explanation_version,r.generated_at FROM mp_recommendation r JOIN mp_decision_profile p ON p.id=r.decision_profile_id AND p.student_id=r.student_id LEFT JOIN mp_experience_run x ON x.id=r.source_run_id AND x.student_id=r.student_id WHERE r.student_id=? ORDER BY r.id",student)){var item=project(r,"public_id:recommendationId","profile_public_id:decisionProfileId","run_public_id:sourceRunId","recommendation_type:recommendationType","rule_version:ruleVersion","primary_major_code:primaryMajor","tie_detected:tieDetected","tie_break_rule:tieBreakRule","explanation_status:explanationStatus","explanation_version:explanationVersion","generated_at:generatedAt");item.set("completedMajorCodes",node(r.get("completed_major_codes_json")));item.set("alternativeMajors",node(r.get("alternative_order_json")));item.set("reasons",rows("SELECT reason_code,reason_type,major_code,display_text,rule_version FROM mp_recommendation_reason WHERE recommendation_id=? ORDER BY reason_order",new Object[]{n(r,"id")},"reason_code:code","reason_type:reasonType","major_code:majorCode","display_text:text","rule_version:ruleVersion"));item.set("decisions",rows("SELECT public_id,decision_no,decision_type,selected_major_code,override_reason_code,override_text,decided_at FROM mp_recommendation_decision WHERE recommendation_id=? AND student_id=? ORDER BY decision_no",new Object[]{n(r,"id"),student},"public_id:decisionId","decision_no:decisionNo","decision_type:decisionType","selected_major_code:selectedMajorCode","override_reason_code:overrideReasonCode","override_text:overrideText","decided_at:decidedAt"));recommendations.add(item);}out.set("recommendations",recommendations);
        var runs=mapper.createArrayNode();for(var r:db.queryForList("SELECT r.id,r.public_id,r.content_kind,r.run_mode,r.demo_key,m.code AS major_code,r.status,r.attempt_no,r.created_reason,r.current_step_key,r.created_at,r.started_at,r.paused_at,r.task_completed_at,r.reflection_completed_at,r.closed_at,r.replaced_at,t.public_id AS task_public_id,t.version_code,t.manifest_json,p.public_id AS profile_public_id,o.public_id AS origin_public_id,rec.public_id AS recommendation_public_id,d.public_id AS decision_public_id FROM mp_experience_run r LEFT JOIN mp_major m ON m.id=r.major_id LEFT JOIN mp_task_version t ON t.id=r.task_version_id LEFT JOIN mp_decision_profile p ON p.id=r.decision_profile_id AND p.student_id=r.student_id LEFT JOIN mp_experience_run o ON o.id=r.origin_run_id AND o.student_id=r.student_id LEFT JOIN mp_recommendation rec ON rec.id=r.source_recommendation_id AND rec.student_id=r.student_id LEFT JOIN mp_recommendation_decision d ON d.id=r.source_decision_id AND d.student_id=r.student_id WHERE r.student_id=? ORDER BY r.id",student))runs.add(run(r,student));out.set("runs",runs);
        var feedback=mapper.createArrayNode();for(var r:db.queryForList("SELECT f.public_id,f.feedback_type,f.context_type,f.rating,f.comment_text,f.survey_version,f.answers_json,f.diagnostic_code,f.created_at,r.public_id AS run_public_id,p.public_id AS report_public_id,v.public_id AS version_public_id FROM mp_student_feedback f LEFT JOIN mp_experience_run r ON r.id=f.experience_run_id AND r.student_id=f.student_id LEFT JOIN mp_report p ON p.id=f.report_id AND p.student_id=f.student_id LEFT JOIN mp_report_version v ON v.id=f.report_version_id AND v.report_id=p.id WHERE f.student_id=? ORDER BY f.id",student)){var item=project(r,"public_id:feedbackId","feedback_type:feedbackType","context_type:contextType","rating:rating","comment_text:comment","survey_version:surveyVersion","diagnostic_code:diagnosticCode","created_at:createdAt","run_public_id:runId","report_public_id:reportId","version_public_id:reportVersionId");var answers=node(r.get("answers_json"));if(!answers.isNull())answers=closedFields(answers,Set.of("insightResponse","clarityChange","nextAction"));item.set("answers",answers);feedback.add(item);}out.set("feedback",feedback);return out;
    }
    private ObjectNode run(Map<String,Object> r,long student){
        long run=n(r,"id");var manifest=node(r.get("manifest_json"));var out=project(r,"public_id:runId","content_kind:contentKind","run_mode:runMode","demo_key:demoKey","major_code:majorCode","status:status","attempt_no:attemptNo","created_reason:createdReason","current_step_key:currentStepKey","created_at:createdAt","started_at:startedAt","paused_at:pausedAt","task_completed_at:taskCompletedAt","reflection_completed_at:reflectionCompletedAt","closed_at:closedAt","replaced_at:replacedAt","task_public_id:taskVersionId","version_code:taskVersionCode","profile_public_id:decisionProfileId","origin_public_id:originRunId","recommendation_public_id:sourceRecommendationId","decision_public_id:sourceDecisionId");if(!manifest.isNull())out.put("taskTitle",manifest.path("title").asText());
        if("REVIEW_PREVIEW".equals(s(r,"run_mode"))){
            var proof=ProductEligibility.authorization(db,student,run);
            if(proof!=null){var visible=project(proof,"public_id:authorizationId","original_run_mode:originalRunMode","authority_code:authorityCode","authorized_at:authorizedAt");visible.put("experienceRunId",s(r,"public_id")).put("taskVersionId",s(r,"task_public_id")).put("decisionProfileId",s(r,"profile_public_id"));out.set("productAuthorization",visible);}
        }
        var steps=mapper.createArrayNode();for(var st:db.queryForList("SELECT id,step_key,status,current_revision,first_started_at,last_saved_at,completed_at FROM mp_step_state WHERE experience_run_id=? ORDER BY id",run)){var definition=definition(manifest,"steps","stepKey",s(st,"step_key"));var item=project(st,"step_key:stepKey","status:status","current_revision:currentRevision","first_started_at:firstStartedAt","last_saved_at:lastSavedAt","completed_at:completedAt");item.put("title",definition.path("title").asText());var revisions=mapper.createArrayNode();for(var rev:db.queryForList("SELECT revision_no,response_schema_version,response_json,validation_status,client_saved_at,received_at FROM mp_step_response_revision WHERE step_state_id=? ORDER BY revision_no",n(st,"id"))){var one=project(rev,"revision_no:revision","response_schema_version:responseSchemaVersion","validation_status:validationStatus","client_saved_at:clientSavedAt","received_at:savedAt");one.set("response",validated(definition.path("inputSchema"),node(rev.get("response_json")),false));revisions.add(one);}item.set("revisions",revisions);steps.add(item);}out.set("steps",steps);
        var executions=mapper.createArrayNode();for(var e:db.queryForList("SELECT public_id,step_key,engine_type,engine_version,operation_key,input_revision,trust_level,status,result_json,error_code,duration_ms,source_revisions_json,created_at FROM mp_execution_record WHERE experience_run_id=? ORDER BY id",run)){var item=project(e,"public_id:executionId","step_key:stepKey","engine_type:engineType","engine_version:engineVersion","operation_key:operationKey","input_revision:inputRevision","trust_level:trustLevel","status:status","error_code:errorCode","duration_ms:durationMs","created_at:createdAt");item.set("sourceRevisions",sourceVector(node(e.get("source_revisions_json")),manifest));if("SUCCEEDED".equals(s(e,"status"))&&"SERVER_VERIFIED".equals(s(e,"trust_level"))){var op=definition(manifest,"operations","operationKey",s(e,"operation_key"));item.set("result",validated(op.path("resultSchema"),node(e.get("result_json")),true));}else item.putNull("result");executions.add(item);}out.set("executions",executions);
        out.set("stepCompletions",rows("SELECT s.step_key,c.response_revision,c.completed_at FROM mp_step_completion c JOIN mp_step_state s ON s.id=c.step_state_id WHERE s.experience_run_id=? ORDER BY c.id",new Object[]{run},"step_key:stepKey","response_revision:responseRevision","completed_at:completedAt"));
        var applications=mapper.createArrayNode();for(var a:db.queryForList("SELECT s.step_key,r.revision_no,e.public_id,x.source_revisions_json FROM mp_response_execution x JOIN mp_step_response_revision r ON r.id=x.response_revision_id JOIN mp_step_state s ON s.id=r.step_state_id JOIN mp_execution_record e ON e.id=x.execution_record_id WHERE s.experience_run_id=? AND e.experience_run_id=? ORDER BY r.id,e.id,x.source_binding_sha256",run,run)){var item=project(a,"step_key:stepKey","revision_no:responseRevision","public_id:executionId");item.set("sourceRevisions",sourceVector(node(a.get("source_revisions_json")),manifest));applications.add(item);}out.set("executionApplications",applications);
        var hints=mapper.createArrayNode();for(var h:db.queryForList("SELECT public_id,step_key,hint_level,input_revision,status,response_json,requested_at,completed_at FROM mp_hint_request WHERE experience_run_id=? ORDER BY id",run)){var item=project(h,"public_id:hintId","step_key:stepKey","hint_level:hintLevel","input_revision:inputRevision","status:status","requested_at:requestedAt","completed_at:completedAt");var response=node(h.get("response_json"));if(!response.isNull())response=closedFields(response,Set.of("schemaVersion","taskVersionId","stepKey","hintLevel","hintText"));item.set("response",response);hints.add(item);}out.set("hints",hints);
        var reflections=mapper.createArrayNode();for(var f:db.queryForList("SELECT id,public_id,question_set_version,status,draft_json,follow_up_json,submitted_at,row_version FROM mp_reflection WHERE experience_run_id=? ORDER BY id",run)){var item=project(f,"public_id:reflectionId","question_set_version:questionSetVersion","status:status","submitted_at:submittedAt","row_version:revision");item.set("answers",validated(manifest.path("reflection").path("inputSchema"),node(f.get("draft_json")),false));var follow=node(f.get("follow_up_json"));var visible=mapper.createObjectNode();if(!follow.isNull()){visible.put("status",follow.path("status").asText());var response=follow.path("response");if(response.isObject())visible.set("response",closedFields(response,Set.of("schemaVersion","taskVersionId","questionSetVersion","question")));else visible.putNull("response");}item.set("followUp",visible);item.set("submittedAnswers",rowsWithAnswers(n(f,"id")));reflections.add(item);}out.set("reflections",reflections);return out;
    }
    private ArrayNode rowsWithAnswers(long reflection){var out=mapper.createArrayNode();for(var r:db.queryForList("SELECT question_key,answer_json,answer_source,ai_question_text,created_at FROM mp_reflection_answer WHERE reflection_id=? ORDER BY id",reflection)){var item=project(r,"question_key:questionKey","answer_source:answerSource","ai_question_text:aiQuestionText","created_at:answeredAt");item.set("answer",node(r.get("answer_json")));out.add(item);}return out;}

    private List<Published> publishedReports(long student){var out=new ArrayList<Published>();for(var r:db.queryForList("SELECT p.public_id AS report_public_id,v.public_id,v.version_no,v.status,v.structured_report_json,v.report_body_sha256 FROM mp_report_version v JOIN mp_report p ON p.id=v.report_id WHERE p.student_id=? AND v.status IN ('READY','SUPERSEDED') ORDER BY p.id,v.version_no",student)){var body=node(r.get("structured_report_json"));if(!reportSchema.output(body)||!CanonicalJson.hash(mapper,body).equals(s(r,"report_body_sha256"))||!s(r,"report_public_id").equals(body.path("reportId").asText())||!s(r,"public_id").equals(body.path("reportVersionId").asText())||n(r,"version_no")!=body.path("versionNo").asLong())throw invalid();for(var source:body.path("sources")){if(db.queryForObject("SELECT COUNT(*) FROM mp_experience_run r JOIN mp_task_version t ON t.id=r.task_version_id WHERE r.public_id=? AND r.student_id=? AND t.public_id=?",Integer.class,source.path("experienceRunId").asText(),student,source.path("taskVersionId").asText())!=1)throw invalid();var ref=source.path("sourceReport");if(!ref.isNull()&&!ref.isMissingNode()&&db.queryForObject("SELECT COUNT(*) FROM mp_report_version v JOIN mp_report p ON p.id=v.report_id WHERE v.public_id=? AND p.public_id=? AND p.student_id=? AND v.status IN ('READY','SUPERSEDED')",Integer.class,ref.path("reportVersionId").asText(),ref.path("reportId").asText(),student)!=1)throw invalid();}out.add(new Published(s(r,"report_public_id"),s(r,"public_id"),n(r,"version_no"),s(r,"status"),body));}return out;}

    static String escape(String value){return value.replace("&","&amp;").replace("<","&lt;").replace(">","&gt;").replace("\"","&quot;").replace("'","&#39;");}
    private static void appendAnalysis(StringBuilder text,JsonNode analysis){
        if(analysis.isMissingNode()||analysis.isNull())return;
        text.append("<section><h2>你的作答分析</h2>");
        if(!"COMPLETE".equals(analysis.path("status").asText())){text.append("<p>本次深入作答分析尚未完成，下方记录不代表已经评价答案质量。</p></section>");return;}
        text.append("<p>").append(escape(analysis.path("summary").asText())).append("</p><p>以下是 AI 对本次作答的解释，可能存在不准确之处，不判断长期能力或专业适合度。</p>");
        for(var major:analysis.path("majorAnalyses")){text.append("<h3>").append(escape(major.path("majorCode").asText())).append("</h3>");for(String field:List.of("preparation","taskReasoning","nextPractice"))text.append("<p>").append(escape(major.path(field).asText())).append("</p>");}
        for(var finding:analysis.path("findings")){text.append("<article><h3>").append(escape(finding.path("stepTitle").asText())).append("</h3><small>").append(escape(finding.path("majorCode").asText())).append(" · ").append(escape(finding.path("assessment").asText())).append("</small><blockquote>").append(escape(finding.path("answerQuote").asText())).append("</blockquote>");for(String field:List.of("reasoning","improvement","example"))text.append("<p>").append(escape(finding.path(field).asText())).append("</p>");text.append("</article>");}
        for(var comparison:analysis.path("crossMajorAnalysis")){text.append("<h3>跨专业观察</h3>");for(String field:List.of("observation","interpretation","nextExperiment"))text.append("<p>").append(escape(comparison.path(field).asText())).append("</p>");}
        text.append("</section>");
    }
    private JsonNode validated(JsonNode schema,JsonNode value,boolean complete){try{TaskInputValidator.validate(schema,value,complete);return value;}catch(RuntimeException invalid){throw invalid();}}
    private JsonNode sourceVector(JsonNode value,JsonNode manifest){if(value.isNull())return mapper.createObjectNode();if(!value.isObject())throw invalid();for(var it=value.fields();it.hasNext();){var entry=it.next();definition(manifest,"steps","stepKey",entry.getKey());if(!entry.getValue().isIntegralNumber()||!entry.getValue().canConvertToLong()||entry.getValue().longValue()<0)throw invalid();}return value;}
    private JsonNode definition(JsonNode manifest,String list,String key,String value){for(var item:manifest.path(list))if(item.path(key).asText().equals(value))return item;throw invalid();}
    private Set<String> questionKeys(){var keys=new HashSet<String>();for(var page:QuestionnaireDefinition.current().pages())for(var field:page.fields())keys.add(field.key());return keys;}
    private JsonNode questionnaireAnswers(JsonNode value){return closedFields(value,questionKeys());}
    private JsonNode closedFields(JsonNode value,Set<String> keys){if(!value.isObject())throw invalid();for(var it=value.fieldNames();it.hasNext();)if(!keys.contains(it.next()))throw invalid();return value;}
    private ArrayNode rows(String sql,Object[] args,String... fields){var out=mapper.createArrayNode();for(var row:db.queryForList(sql,args))out.add(project(row,fields));return out;}
    private ObjectNode project(Map<String,Object> row,String... fields){var out=mapper.createObjectNode();for(String field:fields){var parts=field.split(":",2);Object value=row.get(parts[0]);if(value instanceof Timestamp t)value=t.toInstant().toString();else if(value instanceof java.time.LocalDateTime t)value=t.toInstant(java.time.ZoneOffset.UTC).toString();out.set(parts[1],mapper.valueToTree(value));}return out;}
    private JsonNode node(Object value){try{return value==null?NullNode.getInstance():mapper.readTree(value.toString());}catch(Exception invalid){throw invalid();}}
