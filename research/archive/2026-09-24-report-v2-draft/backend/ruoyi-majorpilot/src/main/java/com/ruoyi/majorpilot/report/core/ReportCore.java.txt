    private static boolean productAuthorized(SourcePins pins,Header header){
        var proof=pins.productAuthorization();
        if("STANDARD".equals(pins.runMode()))return proof==null;
        return "REVIEW_PREVIEW".equals(pins.runMode())&&proof!=null&&id(proof.authorizationId())
            &&"PROJECT_OWNER_REQUEST".equals(proof.authorityCode())&&safe(proof.authorizationReference(),200)
            &&proof.authorizedAt()!=null&&!proof.authorizedAt().isAfter(header.generatedAt())
            &&"REVIEW_PREVIEW".equals(proof.originalRunMode())
            &&Objects.equals(pins.taskManifestSha256(),proof.taskManifestSha256())
            &&Objects.equals(pins.closedInputSha256(),proof.closedInputSha256());
    }
