package com.ruoyi.majorpilot.integration.ai;
import com.ruoyi.majorpilot.report.core.ReportModels.*;
import java.math.BigDecimal;
import java.util.List;

/** One attempt = at most one HTTP request. Durable orchestration owns all retry/repair budgets. */
public interface ReportGateway {
    enum Stage { INITIAL, REPAIR, REVIEW }
    enum Status { PROPOSAL_RECEIVED, INVALID_OUTPUT, UPSTREAM_FAILURE }
    /** Public routing identity only; never includes credentials or a provider endpoint. */
    record Identity(String provider,String model){
        public boolean known(){return knownPart(provider)&&knownPart(model);}
        private static boolean knownPart(String value){return value!=null&&value.matches("[A-Za-z0-9][A-Za-z0-9._-]{0,119}")&&!java.util.Set.of("UNKNOWN","PENDING","DISABLED","UNAVAILABLE").contains(value.toUpperCase(java.util.Locale.ROOT));}
    }
    record Attempt(String invocationId,Stage stage,SafeModelInput input,List<Violation> repairIssues,long remainingBudgetMs){public Attempt{repairIssues=List.copyOf(repairIssues);}}
    record Telemetry(String provider,String model,long durationMs,int httpAttempts,String inputSchemaVersion,String inputSha256,List<String> inputFieldKeys,Long inputTokens,Long outputTokens,BigDecimal estimatedCost,Boolean thinkingEnabled,Long reasoningTokens){
        public Telemetry{inputFieldKeys=List.copyOf(inputFieldKeys);}
        public Telemetry(String provider,String model,long durationMs,int httpAttempts,String inputSchemaVersion,String inputSha256,List<String> inputFieldKeys,Long inputTokens,Long outputTokens,BigDecimal estimatedCost){this(provider,model,durationMs,httpAttempts,inputSchemaVersion,inputSha256,inputFieldKeys,inputTokens,outputTokens,estimatedCost,null,null);}
    }
    record Outcome(Status status,Proposal proposal,List<Violation> violations,String safeErrorCode,Telemetry telemetry){public Outcome{violations=List.copyOf(violations);}}
    boolean available();
    Identity identity();
    Outcome attempt(Attempt request);
}
