    JsonNode build(long student,long run,long profile,TaskContent.Version task,JsonNode closed,List<EvidenceValue> evidence,String rubricVersion){
        requireRubricVersion(rubricVersion);
        var manifest=task.manifest();var result=store.mapper().createObjectNode();result.put("schemaVersion",VERSION);
        var taskContext=select(manifest,"schemaVersion","majorCode","versionCode","title","scenario","goals");
        taskContext.put("assessmentRubricVersion",rubricVersion);
        taskContext.put("assessmentBoundary","只评价本次提交在题目与已验证结果下是否有依据，不推断长期能力、人格、专业适配度或商业收益。提示使用不是负面能力证据，未知学科背景不得臆测。");
        taskContext.put("professionalReviewClaimed",false);
        taskContext.set("materials",materials(task));result.set("task",taskContext);
        var frozen=responses(run,closed);var frozenSources=result.putArray("frozenSources");var steps=result.putArray("steps");
        for(var definition:manifest.path("steps")){
            String key=definition.path("stepKey").asText();var response=frozen.get(key);require(response!=null);var raw=response.value();
            var step=steps.addObject();step.put("stepKey",key);step.put("title",definition.path("title").asText());step.set("instructions",definition.path("instructions").deepCopy());
            step.set("inputSchema",definition.path("inputSchema").deepCopy());
            step.set("rubric",store.mapper().valueToTree(map("version",rubricVersion,"source","TASK_INSTRUCTIONS_AND_PUBLIC_MATERIALS","criteria",criteria(key,rubricVersion))));
            step.set("submittedResponse",schemaProjection(raw,definition.path("inputSchema")));
            var answers=step.putArray("answers");
            var analysis=evidence.stream().filter(e->e.type().equals("STUDENT_ANALYSIS")&&e.key().equals("analysis:"+key)).findFirst().orElse(null);
            if(analysis!=null){require(analysis.allowedClaimTypes().isEmpty()&&analysis.value().path("verificationStatus").asText().equals("UNVERIFIED"));
                var original=analysis.value().path("answers");require(original.isObject());original.fields().forEachRemaining(field->require(raw.has(field.getKey())&&raw.get(field.getKey()).equals(field.getValue())));
                flatten(original,definition.path("inputSchema"),"",analysis.publicId(),answers);
            }
            var all=store.mapper().createArrayNode();flatten(raw,definition.path("inputSchema"),"",null,all);
            Set<String> included=new HashSet<>();answers.forEach(a->included.add(a.path("answerField").asText()));
            for(var answer:all)if(!included.contains(answer.path("answerField").asText())){var value=(ObjectNode)answer.deepCopy();String field=value.path("answerField").asText(),ref="frozen-response:"+task.majorCode()+"/"+key+":"+field;
                value.put("sourceRef",ref);value.put("sourceSha256",response.sha());answers.add(value);
                frozenSources.add(store.mapper().valueToTree(map("sourceRef",ref,"sourceType","STEP_RESPONSE","stepKey",key,"answerField",field,"revision",response.revision(),"sourceSha256",response.sha(),"value",value.path("value").asText())));
            }
            var observations=step.putArray("responseEvidence");for(var e:evidence)if(e.key().equals("step:"+key))observations.add(evidence(e));
        }
        var executions=result.putArray("verifiedExecutions");evidence.stream().filter(e->e.type().equals("VERIFIED_EXECUTION")).forEach(e->executions.add(evidence(e)));
        result.set("reflection",reflection(run,closed,manifest,evidence,frozenSources));
        var support=result.putObject("supportContext");support.put("interpretationBoundary","提示层级说明支持条件，不等同于能力高低；没有提示记录也不能推断独立完成或既有知识。");
        support.set("hints",hints(run,closed,evidence));
        result.set("background",background(student,profile));
        enrichAggregateLineage(store.mapper(),result,manifest,rubricVersion);
        if(RUBRIC.equals(rubricVersion)&&"AI".equals(task.majorCode())){
            var assets=new LinkedHashMap<String,byte[]>();for(String key:ReportAiAnalysisContext.ASSETS)assets.put(key,content.publicAsset(task,key).bytes());
            verifyAiExecution(run,task,closed,frozen,evidence,assets.get("classifier-model"));
            ReportAiAnalysisContext.enrich(store.mapper(),result,manifest,assets);
        }
        return result;
    }
    /** Versioned arithmetic from pinned public rows, never a grade or an authored answer key. */
    static void enrichAggregateLineage(ObjectMapper mapper,ObjectNode context,JsonNode manifest,String rubricVersion){
        requireRubricVersion(rubricVersion);
        if((FRAME_RUBRIC.equals(rubricVersion)||TARGET_RUBRIC.equals(rubricVersion)||RUBRIC.equals(rubricVersion))&&"DATA_SCIENCE".equals(manifest.path("majorCode").asText())){
            require(VERSION.equals(context.path("schemaVersion").asText())&&rubricVersion.equals(context.at("/task/assessmentRubricVersion").asText()));
            ReportDsAnalysisContext.enrich(mapper,context,manifest);return;
        }
        if(!(AGGREGATE_RUBRIC.equals(rubricVersion)||FRAME_RUBRIC.equals(rubricVersion)||TARGET_RUBRIC.equals(rubricVersion)||RUBRIC.equals(rubricVersion))||!"BUSINESS_ANALYTICS".equals(manifest.path("majorCode").asText()))return;
        require(VERSION.equals(context.path("schemaVersion").asText())&&"task-manifest-v4".equals(manifest.path("schemaVersion").asText()));
        var task=context.path("task");require(task.isObject()&&task.path("schemaVersion").equals(manifest.path("schemaVersion"))&&task.path("majorCode").equals(manifest.path("majorCode"))&&task.path("versionCode").equals(manifest.path("versionCode"))&&rubricVersion.equals(task.path("assessmentRubricVersion").asText()));
        JsonNode rows=uniqueEntry(task.path("materials"),"assetKey","weekly-funnel").path("content");
        JsonNode dictionary=uniqueEntry(task.path("materials"),"assetKey","data-dictionary").path("content");
        require("ba-dictionary-v1".equals(dictionary.path("schemaVersion").asText())&&"COHORT_COURSE_SOURCE".equals(dictionary.path("observationUnit").asText()));
        for(String dimension:List.of("course_track","signup_source"))require(dictionary.path("groupOrder").path(dimension).equals(mapper.valueToTree(BaSemantics.GROUPS.get(dimension))));
        String datasetSha=uniqueEntry(manifest.path("assets"),"assetKey","weekly-funnel").path("sha256").asText();require(datasetSha.matches("[a-f0-9]{64}"));
        var operation=uniqueEntry(manifest.path("operations"),"operationKey","ba-segments");
        require("BUSINESS_ANALYSIS".equals(operation.path("engineType").asText())&&"ba-metrics-v1".equals(operation.path("engineVersion").asText())&&"weekly-funnel".equals(operation.path("datasetAssetKey").asText()));
        var b2=uniqueEntry(context.path("steps"),"stepKey","b2-metric").path("submittedResponse");
        var b3=uniqueEntry(context.path("steps"),"stepKey","b3-funnel").path("submittedResponse");
        var b4=uniqueEntry(context.path("steps"),"stepKey","b4-segment").path("submittedResponse");
        var parameters=mapper.createObjectNode();for(String key:BaSemantics.WINDOWS)parameters.set(key,b2.path(key));
        parameters.set("numerator",b2.path("numerator"));parameters.set("denominator",b2.path("denominator"));parameters.set("focusTransitionKey",b3.path("focusTransitionKey"));parameters.set("segmentBy",b4.path("segmentBy"));
        final JsonNode selected;
        try{selected=BaStatistics.calculate("ba-segments",rows,parameters,datasetSha);}catch(RuntimeException invalid){throw new com.ruoyi.majorpilot.shared.ApiException(409,"REPORT_SOURCE_INVALID");}
        var refs=mapper.createArrayNode();
        for(var execution:context.path("verifiedExecutions"))if("VERIFIED_EXECUTION".equals(execution.path("type").asText())&&sameNumbers(selected,execution.path("value")))refs.add(execution.path("evidenceId"));
        require(!refs.isEmpty()); // No guessed windows, mixed material versions, or unbound B4 results.
        var facts=mapper.createObjectNode().put("schemaVersion","ba-aggregate-lineage-v1").put("sourceAssetKey","weekly-funnel").put("datasetSha256",datasetSha).put("materialCanonicalSha256",CanonicalJson.hash(mapper,rows));
        facts.set("selection",parameters.deepCopy());facts.put("observationWindowDays",28);facts.put("overallEqualsDisjointGroupCountSums",true);
        var lineage=facts.putObject("verifiedSegmentSource").put("stepKey","b4-segment").put("operationKey","ba-segments").put("segmentBy",parameters.path("segmentBy").asText());lineage.set("evidenceRefs",refs);
        facts.putArray("aggregationRules")
            .add("同一数据版本、窗口及分母口径下，课程分组或来源分组各自互斥且穷尽；先加总分子和分母计数再算整体率，不平均分组百分比。B4的完整分组计数也能支持可复算的整体观察，不因B2展示更直接就把B4引用判错；未写加总过程可建议补充表达。")
            .add("每组未启动人数=报名人数-首课人数。课程、渠道及课程×渠道的未启动分布直接从已有交叉聚合行计算，不是仍需采集的信息；不能只凭两张边际分布猜交叉分布。")
            .add("支持联系为事件次数，可加总；每100名报名者支持联系=合计联系次数/合计报名人数×100，不是被联系人数百分比。计数、比例及构成不能单独证明原因或干预效果。");
        facts.putArray("notRecordedInAggregateRows").add("个体未启动的具体障碍和原因").add("支持信息是否触达某人、内容是否被理解").add("个体支持事件与启动结果的关联").add("未来干预的反事实效果、资源可行性与统计充分性");
        var periods=facts.putObject("periods");
        var grouped=new LinkedHashMap<String,JsonNode>();
        for(String dimension:List.of("course_track","signup_source")){var p=parameters.deepCopy();p.put("segmentBy",dimension);grouped.put(dimension,BaStatistics.calculate("ba-segments",rows,p,datasetSha));}
        for(String period:List.of("baseline","comparison")){
            int start=parameters.path(period+"StartWeek").asInt(),end=parameters.path(period+"EndWeek").asInt();var out=periods.putObject(period).put("startWeek",start).put("endWeek",end);var overall=out.putObject("overall");
            for(String key:List.of("enrolled_count","first_lesson_count","completed_count","support_contact_count","nonStarterCount"))overall.put(key,0L);
            for(String dimension:List.of("course_track","signup_source")){
                var groups=out.putObject(dimension.equals("course_track")?"byCourse":"bySource");
                for(var segment:grouped.get(dimension).path("segments")){
                    var p=segment.path(period);var counts=p.path("stageCounts");var group=groups.putObject(segment.path("segmentKey").asText());
                    for(String key:List.of("enrolled_count","first_lesson_count","completed_count"))group.set(key,counts.path(key));
                    group.set("support_contact_count",p.path("supportContactCount"));group.put("nonStarterCount",counts.path("enrolled_count").asLong()-counts.path("first_lesson_count").asLong());
                    if(dimension.equals("course_track"))group.fields().forEachRemaining(field->overall.put(field.getKey(),overall.path(field.getKey()).asLong()+field.getValue().asLong()));
                }
            }
            var cross=out.putObject("nonStartersByCourseAndSource");
            for(String course:BaSemantics.GROUPS.get("course_track")){var channels=cross.putObject(course);for(String channel:BaSemantics.GROUPS.get("signup_source")){
                long count=0;for(var row:rows)if(row.path("week").asInt()>=start&&row.path("week").asInt()<=end&&course.equals(row.path("course_track").asText())&&channel.equals(row.path("signup_source").asText()))count+=row.path("enrolled_count").asLong()-row.path("first_lesson_count").asLong();channels.put(channel,count);
            }}
        }
        // v2.2 must retain its exact tree, including the absence of the fields below.
        if(FRAME_RUBRIC.equals(rubricVersion)||TARGET_RUBRIC.equals(rubricVersion)||RUBRIC.equals(rubricVersion))enrichCrossGroupInventory(facts,rows,parameters);
        ((ObjectNode)task).set("aggregateLineage",facts);
        if(TARGET_RUBRIC.equals(rubricVersion)||RUBRIC.equals(rubricVersion))((ObjectNode)task).set("actionTargetFrame",ReportBaActionTargetContext.build(mapper,uniqueEntry(context.path("steps"),"stepKey","b6-action").path("submittedResponse"),parameters,b4.path("prioritySegmentKey").asText(),selected,refs));
    }
    /** Cross counts originate in actual joint rows, not inferred from two marginal tables. */
    private static void enrichCrossGroupInventory(ObjectNode facts,JsonNode rows,JsonNode parameters){
        facts.put("schemaVersion","ba-aggregate-lineage-v2");
        var source=facts.putObject("crossGroupSource").put("kind","PINNED_PUBLIC_ROWS").put("sourceAssetKey","weekly-funnel");
        source.set("datasetSha256",facts.path("datasetSha256"));source.set("materialCanonicalSha256",facts.path("materialCanonicalSha256"));
        source.putArray("groupBy").add("course_track").add("signup_source");
        var windows=source.putObject("windows");for(String key:BaSemantics.WINDOWS)windows.set(key,parameters.path(key));
        source.put("aggregation","SUM_COUNTS_WITHIN_WINDOW_AND_JOINT_GROUP").put("nonStarterDefinition","enrolled_count - first_lesson_count");
        source.put("deltaDefinition","COMPARISON_MINUS_BASELINE").put("interpretationBoundary","DESCRIPTIVE_NOT_CAUSAL");
        var keys=List.of("enrolled_count","first_lesson_count","completed_count","support_contact_count","nonStarterCount");
        var cells=facts.putArray("crossGroupCounts");
        for(String course:BaSemantics.GROUPS.get("course_track"))for(String channel:BaSemantics.GROUPS.get("signup_source")){
            var cell=cells.addObject().put("course_track",course).put("signup_source",channel);
            for(String period:List.of("baseline","comparison")){
                int start=parameters.path(period+"StartWeek").asInt(),end=parameters.path(period+"EndWeek").asInt();var counts=cell.putObject(period);keys.forEach(key->counts.put(key,0L));
                for(var row:rows)if(row.path("week").asInt()>=start&&row.path("week").asInt()<=end&&course.equals(row.path("course_track").asText())&&channel.equals(row.path("signup_source").asText()))
                    for(String key:keys)if(!key.equals("nonStarterCount"))counts.put(key,counts.path(key).asLong()+row.path(key).asLong());
                counts.put("nonStarterCount",counts.path("enrolled_count").asLong()-counts.path("first_lesson_count").asLong());
            }
            var delta=cell.putObject("delta");for(String key:keys)delta.put(key,cell.path("comparison").path(key).asLong()-cell.path("baseline").path(key).asLong());
        }
        var inventory=facts.putObject("informationInventory").put("schemaVersion","ba-information-inventory-v1");var available=inventory.putArray("availableDerived");
        available.addObject().put("description","固定窗口内总体、课程和渠道各自的报名、首课、完成、支持联系及未启动计数").put("factsPath","/task/aggregateLineage/periods");
        available.addObject().put("description","固定窗口内课程×渠道交叉分组的报名、首课、完成、支持联系及未启动计数；可计算这些已观察人数的渠道构成").put("factsPath","/task/aggregateLineage/crossGroupCounts");
        available.addObject().put("description","两窗口各课程×渠道计数的增减及其对总体人数变化的算术贡献；这是描述性构成，不是原因归因").put("factsPath","/task/aggregateLineage/crossGroupCounts");
        inventory.set("unobserved",facts.path("notRecordedInAggregateRows").deepCopy());
    }
    private static JsonNode uniqueEntry(JsonNode array,String key,String value){require(array.isArray());JsonNode found=null;for(var entry:array)if(value.equals(entry.path(key).asText())){require(found==null);found=entry;}require(found!=null);return found;}
    private static boolean sameNumbers(JsonNode a,JsonNode b){return a.equals((left,right)->left.isNumber()&&right.isNumber()?left.decimalValue().compareTo(right.decimalValue()):left.equals(right)?0:1,b);}

    private void verifyAiExecution(long run,TaskContent.Version task,JsonNode closed,Map<String,FrozenResponse> responses,List<EvidenceValue> evidence,byte[] modelBytes){
        final JsonNode model;try{model=store.mapper().readTree(modelBytes);}catch(Exception bad){throw new com.ruoyi.majorpilot.shared.ApiException(409,"REPORT_SOURCE_INVALID");}
        var config=store.mapper().createObjectNode();var a4=responses.get("a4-configure");var a5=responses.get("a5-evaluate");require(a4!=null&&a5!=null);
        for(String key:ReportAiAnalysisContext.CONFIG)config.set(key,a4.value().path(key));
        final JsonNode normalized;try{normalized=com.ruoyi.majorpilot.classifier.FrozenClassifier.normalize(model,config);}catch(RuntimeException bad){throw new com.ruoyi.majorpilot.shared.ApiException(409,"REPORT_SOURCE_INVALID");}
        var sorted=new TreeMap<String,JsonNode>();normalized.fields().forEachRemaining(e->sorted.put(e.getKey(),e.getValue()));
        String configSha=com.ruoyi.majorpilot.shared.Secrets.hash(task.taskVersionId()+"\0ai-evaluate\0"+store.json(sorted));
        require(closed.path("executions").size()==1);var execution=uniqueEntry(closed.path("executions"),"operationKey","ai-evaluate");
        require("a5-evaluate".equals(execution.path("stepKey").asText())&&execution.path("appliedRevision").asLong()==a5.revision());
        var sources=store.mapper().valueToTree(Map.of("a4-configure",a4.revision()));require(store.hash(sources).equals(store.hash(execution.path("sourceRevisions"))));
        var row=store.db().queryForMap("SELECT public_id,operation_key,step_key,engine_type,engine_version,evaluation_config_sha256,result_json,status,trust_level FROM mp_execution_record WHERE id=? AND experience_run_id=?",execution.path("executionId").asLong(),run);
        var value=store.node(row.get("result_json"));String resultSha=store.hash(value);
        require(s(row,"public_id").equals(execution.path("publicId").asText())&&"ai-evaluate".equals(s(row,"operation_key"))&&"a5-evaluate".equals(s(row,"step_key"))&&"FROZEN_CLASSIFIER".equals(s(row,"engine_type"))&&"classifier-v1".equals(s(row,"engine_version"))&&"SUCCEEDED".equals(s(row,"status"))&&"SERVER_VERIFIED".equals(s(row,"trust_level"))&&configSha.equals(s(row,"evaluation_config_sha256"))&&resultSha.equals(execution.path("resultSha256").asText()));
        require(evidence.stream().filter(e->"VERIFIED_EXECUTION".equals(e.type())).count()==1&&evidence.stream().anyMatch(e->"VERIFIED_EXECUTION".equals(e.type())&&e.key().equals("execution:"+s(row,"public_id"))&&store.hash(e.value()).equals(resultSha)));

AND s.experience_run_id=? AND s.step_key='a5-evaluate' AND r.revision_no=? AND x.source_binding_sha256=?",Integer.class,execution.path("executionId").asLong(),run,a5.revision(),store.hash(sources))>0);
    }
    private Map<String,FrozenResponse> responses(long run,JsonNode closed){
        var result=new LinkedHashMap<String,FrozenResponse>();for(var step:closed.path("steps")){
            var row=store.db().queryForMap("SELECT r.response_json,r.content_sha256,r.revision_no,s.step_key FROM mp_step_response_revision r JOIN mp_step_state s ON s.id=r.step_state_id WHERE r.id=? AND s.experience_run_id=?",step.path("revisionId").asLong(),run);
            var value=store.node(row.get("response_json"));require(s(row,"content_sha256").equals(step.path("sha256").asText())&&store.hash(value).equals(step.path("valueSha256").asText())&&n(row,"revision_no")==step.path("revision").asLong()&&s(row,"step_key").equals(step.path("stepKey").asText()));
            require(result.put(step.path("stepKey").asText(),new FrozenResponse(value,n(row,"revision_no"),step.path("valueSha256").asText()))==null);
        }return result;
    }

    private JsonNode background(long student,long profile){
        var result=store.mapper().createObjectNode();result.put("attribution","SELF_REPORTED_INTAKE_NOT_VERIFIED_ABILITY");
        var allowed=Set.of("currentField","consideredMajors","majorFamiliarity","foundationSelfView","scenarioPreference","workElements","decisionUncertainties");
        for(var answer:store.db().queryForList("SELECT a.question_key,a.answer_json FROM mp_decision_answer a JOIN mp_decision_profile p ON p.id=a.decision_profile_id WHERE p.id=? AND p.student_id=?         return result;
    }
    private JsonNode hints(long run,JsonNode closed,List<EvidenceValue> evidence){
        var result=store.mapper().createArrayNode();for(var frozen:closed.path("hints")){
