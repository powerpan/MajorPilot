    @Override public Outcome attempt(Attempt r){long start=System.nanoTime();String inputSha=null;int attempts=0;Long in=null,out=null,reasoning=null;try{
        if(r==null||r.input()==null||r.stage()==null||r.invocationId()==null||!r.invocationId().matches("[A-Za-z0-9_-]{1,80}")||r.remainingBudgetMs()<1||r.remainingBudgetMs()>ReportModelSettings.MAX_TOTAL_BUDGET_MS||r.repairIssues().size()>20||!Set.of("report-model-input-v1","report-model-input-v2","report-model-review-input-v1",ReportBindingCodec.INPUT_SCHEMA,ReportBindingCodec.REVIEW_SCHEMA).contains(r.input().schemaVersion()))throw new AiFailure(AiFailure.Code.INVALID_INPUT);
        boolean review=isReview(r.input());
        if(review!=(r.stage()==Stage.REVIEW)||review&&!r.repairIssues().isEmpty())throw new AiFailure(AiFailure.Code.INVALID_INPUT);
        var context=r.input().content();if(isBound(r.input()))validateBoundContext(r.input());else if(review)validateReviewContext(context);else if("report-model-input-v2".equals(r.input().schemaVersion()))validateAnalysisContext(context);else validateSafeContext(context);if(r.input().catalogSha256()==null||!r.input().catalogSha256().matches("[a-f0-9]{64}"))throw new AiFailure(AiFailure.Code.INVALID_INPUT);
        for(var issue:r.repairIssues())if(!issue.code().matches("[A-Z_]{1,80}")||!issue.path().matches("/[A-Za-z0-9/]*"))throw new AiFailure(AiFailure.Code.INVALID_INPUT);
        boolean streaming=review||isBound(r.input())||"report-model-input-v2".equals(r.input().schemaVersion());inputSha=CanonicalJson.hash(mapper,context);long timeout=Math.min(settings.requestTimeoutMs(),r.remainingBudgetMs());var builder=HttpRequest.newBuilder(endpoint).timeout(Duration.ofMillis(timeout)).header("Authorization","Bearer "+key).header("Content-Type","application/json");if(streaming)builder.header("Accept","text/event-stream");var request=builder.POST(HttpRequest.BodyPublishers.ofString(body(r))).build();attempts=1;var raw=transport.send(request,timeout);
        if(raw.status()<200||raw.status()>=300)throw new AiFailure(raw.status()==429?AiFailure.Code.RATE_LIMITED:raw.status()==401||raw.status()==403?AiFailure.Code.AUTH_FAILED:raw.status()>=500?AiFailure.Code.UPSTREAM_UNAVAILABLE:AiFailure.Code.UPSTREAM_REJECTED);
        if(raw.body()==null||raw.body().length>(streaming?ReportHttpTransport.maxEnvelopeBytes(settings.maxResponseBytes()):settings.maxResponseBytes()))throw new AiFailure(AiFailure.Code.OUTPUT_LIMIT);JsonNode envelope=mapper.reader().with(com.fasterxml.jackson.core.JsonParser.Feature.STRICT_DUPLICATE_DETECTION).with(DeserializationFeature.FAIL_ON_TRAILING_TOKENS).readTree(raw.body());in=tokens(envelope,"prompt_tokens");out=tokens(envelope,"completion_tokens");reasoning=nonnegativeLong(envelope.at("/usage/completion_tokens_details/reasoning_tokens"));var choice=envelope.path("choices").path(0);if("length".equals(choice.path("finish_reason").asText()))throw new AiFailure(AiFailure.Code.OUTPUT_LIMIT);if(!choice.path("finish_reason").asText().equals("stop")||!choice.at("/message/content").isTextual())throw new AiFailure(AiFailure.Code.INVALID_OUTPUT);if(streaming&&choice.at("/message/content").asText().getBytes(StandardCharsets.UTF_8).length>settings.maxResponseBytes())throw new AiFailure(AiFailure.Code.OUTPUT_LIMIT);
        // Reasoning is never a report, repair transcript, persisted artifact, or logged content.
        byte[] finalContent=choice.at("/message/content").asText().getBytes(StandardCharsets.UTF_8);var proposal=core.parseModelProposal(finalContent,r.input());return new Outcome(Status.PROPOSAL_RECEIVED,proposal,List.of(),null,telemetry(r,start,attempts,inputSha,in,out,reasoning));
    }catch(ProposalException failure){return new Outcome(Status.INVALID_OUTPUT,null,failure.issues,"REPORT_PROPOSAL_INVALID",telemetry(r,start,attempts,inputSha,in,out,reasoning));}catch(AiFailure failure){var invalid=Set.of(AiFailure.Code.INVALID_OUTPUT,AiFailure.Code.OUTPUT_LIMIT).contains(failure.code());return new Outcome(invalid?Status.INVALID_OUTPUT:Status.UPSTREAM_FAILURE,null,invalid?List.of(new ReportModels.Violation("STRUCTURE_INVALID","/")):List.of(),failure.code().name(),telemetry(r,start,attempts,inputSha,in,out,reasoning));}catch(Exception ignored){return new Outcome(Status.INVALID_OUTPUT,null,List.of(new ReportModels.Violation("STRUCTURE_INVALID","/")),"INVALID_OUTPUT",telemetry(r,start,attempts,inputSha,in,out,reasoning));}}
    private Telemetry telemetry(Attempt r,long start,int attempts,String sha,Long in,Long out,Long reasoning){String schema=r!=null&&r.input()!=null?r.input().schemaVersion():"report-model-input-v1";var fields=Set.of("report-model-review-input-v1",ReportBindingCodec.REVIEW_SCHEMA).contains(schema)?List.of("originalInput","untrustedDraft"):ReportBindingCodec.INPUT_SCHEMA.equals(schema)?List.of("reportType","sources","analysisContract","nextRecommendation","bindingCatalog"):"report-model-input-v2".equals(schema)?List.of("reportType","sources","analysisContract","nextRecommendation"):List.of("reportType","requiredSectionOrder","statements","completeValidProposal");return new Telemetry(provider,model,TimeUnit.NANOSECONDS.toMillis(System.nanoTime()-start),attempts,schema,sha,fields,in,out,null,attempts>0?true:null,reasoning);}
    private String body(Attempt r)throws Exception{
        boolean review=r.stage()==Stage.REVIEW;boolean streaming=review||isBound(r.input())||"report-model-input-v2".equals(r.input().schemaVersion());var body=mapper.createObjectNode();body.put("model",model).put("stream",streaming).put("max_tokens",settings.maxOutputTokens());if(streaming)body.putObject("stream_options").put("include_usage",true);
        if(provider.equals("BAILIAN"))body.put("enable_thinking",true).put("thinking_budget",settings.thinkingBudgetTokens());
        else {body.set("thinking",mapper.valueToTree(Map.of("type","enabled")));body.put("reasoning_effort","max");}
        var format=body.putObject("response_format");
        if(streaming&&settings.outputFormat()==ReportModelSettings.OutputFormat.JSON_SCHEMA){
            format.put("type","json_schema");var schema=format.putObject("json_schema");
            schema.put("name",isBound(r.input())?"report_analysis_wire_proposal_v3":"report_analysis_proposal_v2").put("strict",true);
            schema.set("schema",core.providerProposalSchema(r.input()));
        }else{
            // Historical v1 keeps its JSON-object contract; this is not a provider-error fallback.
            format.put("type",settings.outputFormat()==ReportModelSettings.OutputFormat.TEXT_JSON?"text":"json_object");
        }
        String prompt=isBound(r.input())?boundPrompt(review):streaming?ANALYSIS_PROMPT+(review?REVIEW_PROMPT:"")+analysisTypePrompt((review?r.input().content().path("originalInput"):r.input().content()).path("reportType").asText()):LEGACY_PROMPT;
        if(r.stage()==Stage.REPAIR)prompt+="这是同一版本唯一一次修复。重新检查原始证据及任务要求，按问题代码修复结构、引用或分析覆盖，不能杜撰答案或降低分析深度。安全问题代码："+mapper.writeValueAsString(r.repairIssues());
        body.set("messages",mapper.valueToTree(List.of(Map.of("role","system","content",prompt),Map.of("role","user","content",mapper.writeValueAsString(r.input().content())))));
        return mapper.writeValueAsString(body);
    }
    private static boolean isBound(ReportModels.SafeModelInput input){return Set.of(ReportBindingCodec.INPUT_SCHEMA,ReportBindingCodec.REVIEW_SCHEMA).contains(input.schemaVersion());}
    private static boolean isReview(ReportModels.SafeModelInput input){return Set.of("report-model-review-input-v1",ReportBindingCodec.REVIEW_SCHEMA).contains(input.schemaVersion());}
    private void validateBoundContext(ReportModels.SafeModelInput input){
        if(input.content().toString().getBytes(StandardCharsets.UTF_8).length>(isReview(input)?ReportModelSettings.MAX_REVIEW_INPUT_BYTES:ReportModelSettings.MAX_INPUT_BYTES))throw new AiFailure(AiFailure.Code.INVALID_INPUT);
        try{core.validateModelBinding(input);}catch(RuntimeException invalid){throw new AiFailure(AiFailure.Code.INVALID_INPUT);}
    }

    }
