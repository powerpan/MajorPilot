package majorpilot.harness;

import com.fasterxml.jackson.databind.JsonNode;
import com.fasterxml.jackson.databind.ObjectMapper;
import com.ruoyi.majorpilot.integration.ai.ReportGateway;
import com.ruoyi.majorpilot.report.core.ReportBindingCodec;
import com.ruoyi.majorpilot.report.core.ReportCore;
import com.ruoyi.majorpilot.report.core.ReportModels.Violation;
import java.nio.file.Files;
import java.nio.file.Path;
import java.nio.file.LinkOption;
import java.util.List;
import java.util.Set;
import org.springframework.context.annotation.Primary;
import org.springframework.context.annotation.Profile;
import org.springframework.core.env.Environment;
import org.springframework.stereotype.Component;

/** Test-only replay/fault gateway. Intentionally no HTTP client or provider credentials. */
@Component @Primary @Profile("p6-harness")
public final class P6HarnessGateway implements ReportGateway {
    private final ObjectMapper mapper;
    private final ReportCore core;
    private final Path directory;
    public P6HarnessGateway(ObjectMapper mapper, ReportCore core, Environment environment) {
        this.mapper=mapper; this.core=core;
        directory=Path.of(environment.getRequiredProperty("MP_P6_HARNESS_DIR")).toAbsolutePath().normalize();
        if (!directory.endsWith("data/p6-harness")) throw new IllegalStateException("Harness control path rejected");
    }
    @Override public boolean available(){ return !mode().equals("AI_OFF"); }
    @Override public Identity identity(){return new Identity("TEST_REPLAY","test-report-analysis-v2");}
    private String mode() {
        try {
            Path file=directory.resolve("gateway-control.json");
            if(!Files.isRegularFile(file,LinkOption.NOFOLLOW_LINKS)||Files.size(file)>512)throw new IllegalStateException();
            var control=mapper.readTree(Files.readString(file));
            if(!control.isObject()||control.size()!=1||!control.path("mode").isTextual())throw new IllegalStateException();
            String value=control.path("mode").asText();
            if(!Set.of("PASS","AI_OFF","INVALID_INITIAL","INVALID_ALWAYS","INVALID_EXPERIENCE_ALWAYS","UPSTREAM_429","TIMEOUT","HOLD_INITIAL","HOLD_REPAIR","HOLD_REVIEW").contains(value))throw new IllegalStateException();
            return value;
        }catch(Exception failure){throw new IllegalStateException("Harness control rejected; payload suppressed");}
    }
    @Override public Outcome attempt(Attempt request) {
        long start=System.nanoTime();
        try {
            String mode=mode();boolean holdTimedOut=false;
            if(mode.equals("AI_OFF"))throw new IllegalStateException("Disabled harness gateway");
            if (mode.equals("HOLD_"+request.stage().name())) {
                Path entered=directory.resolve("gateway-entered.json");
                if(Files.isSymbolicLink(entered))throw new IllegalStateException("Harness control rejected");
                Files.writeString(entered, mapper.writeValueAsString(java.util.Map.of(
                    "stage", request.stage().name(), "invocationId", request.invocationId())));
                long deadline=System.nanoTime()+Math.min(request.remainingBudgetMs(),45000L)*1_000_000L;
                while (!Files.isRegularFile(directory.resolve("gateway-release"),LinkOption.NOFOLLOW_LINKS) && System.nanoTime()<deadline) Thread.sleep(50);
                holdTimedOut=System.nanoTime()>=deadline;
            }
            // One simulated transport attempt; TEST_REPLAY is never real provider traffic or model-quality evidence.
            boolean bound=Set.of(ReportBindingCodec.INPUT_SCHEMA,ReportBindingCodec.REVIEW_SCHEMA).contains(request.input().schemaVersion());
            boolean review="report-model-review-input-v1".equals(request.input().schemaVersion())||ReportBindingCodec.REVIEW_SCHEMA.equals(request.input().schemaVersion());
            boolean analysis=bound||review||"report-model-input-v2".equals(request.input().schemaVersion());
            if(!analysis&&!"report-model-input-v1".equals(request.input().schemaVersion()))throw new IllegalStateException("Unknown harness report input");
            if(review!=(request.stage()==Stage.REVIEW))throw new IllegalStateException("Harness input stage mismatch");
            if(bound)core.validateModelBinding(request.input());
            var originalInput=review?request.input().content().path("originalInput"):request.input().content();
            String reportType=(analysis?originalInput:originalInput.path("completeValidProposal")).path("reportType").asText();
            var inputKeys=new java.util.ArrayList<String>();request.input().content().fieldNames().forEachRemaining(inputKeys::add);
            var telemetry=new Telemetry("TEST_REPLAY",analysis?"test-report-analysis-v2":"test-report-proposal-v1",Math.max(0,(System.nanoTime()-start)/1_000_000),1,
                request.input().schemaVersion(),core.hash(request.input().content()),inputKeys,null,null,null);
            // Target an experience fallback without also breaking its automatically queued comparison.
            boolean invalidExperience=mode.equals("INVALID_EXPERIENCE_ALWAYS")&&reportType.equals("EXPERIENCE")&&request.stage()!=Stage.REVIEW;
            if (mode.equals("INVALID_ALWAYS") || invalidExperience || Set.of("INVALID_INITIAL","HOLD_REPAIR").contains(mode)&&request.stage()==Stage.INITIAL)
                return new Outcome(Status.INVALID_OUTPUT,null,List.of(new Violation("TEST_INVALID_PROPOSAL",analysis?"/analysis":"/sections")),"REPORT_OUTPUT_INVALID",telemetry);
            if (mode.equals("UPSTREAM_429") || mode.equals("TIMEOUT") || holdTimedOut)
                return new Outcome(Status.UPSTREAM_FAILURE,null,List.of(),holdTimedOut||mode.equals("TIMEOUT")?"AI_TIMEOUT":"AI_RATE_LIMITED",telemetry);
            // The replay starts from exact local source references, then exercises the declared
            // bound wire codec; it never guesses raw references from provider-side E aliases.
            var decoded=core.parseProposal(mapper.writeValueAsBytes(analysis?syntheticAnalysis(bound?request.input().bindingCatalogContext().analysisContext():originalInput,bound):request.input().content().get("completeValidProposal")));
            if(bound)decoded=core.parseModelProposal(mapper.writeValueAsBytes(core.modelProposal(request.input().bindingCatalogContext(),decoded)),request.input());
            return new Outcome(Status.PROPOSAL_RECEIVED,decoded,List.of(),null,telemetry);
        } catch (InterruptedException interrupted) {
            Thread.currentThread().interrupt(); throw new IllegalStateException("Harness gateway interrupted");
        } catch (Exception failure) {
            throw new IllegalStateException("Harness replay failed; payload suppressed");
        }
    }

    /** Browser-only wiring output with actual synthetic source references; never a real assessment. */
    private JsonNode syntheticAnalysis(JsonNode input,boolean allFields) {
        String explanation="这是浏览器验收专用的合成分析，用于检查本次测试答案的引用、展示和报告流程，不代表真实模型对作答质量的判断。";
        var proposal=mapper.createObjectNode().put("schemaVersion","report-analysis-proposal-v2")
                .put("reportType",input.path("reportType").asText());
        var analysis=proposal.putObject("analysis");analysis.put("summary",explanation);
        var majors=analysis.putArray("majorAnalyses");var findings=analysis.putArray("findings");
        var majorCodes=mapper.createArrayNode();var crossMajorRefs=mapper.createArrayNode();
        for(var source:input.path("sources")) {
            String code=source.path("majorCode").asText(),firstRef=null;majorCodes.add(code);
            for(var step:source.at("/context/steps")) {
                if(step.path("answers").isEmpty())continue;
                for(var answer:step.path("answers")){
                String ref=answer.path(answer.has("evidenceId")?"evidenceId":"sourceRef").asText();
                if(ref.isBlank())throw new IllegalStateException("Harness analysis source reference missing");
                if(firstRef==null)firstRef=ref;
                String original=answer.path("value").asText();
                // A continuous original substring fits the wire limit without changing its wording.
                String quote=original.substring(0,original.offsetByCodePoints(0,Math.min(1200,original.codePointCount(0,original.length()))));
                var finding=findings.addObject().put("majorCode",code).put("stepKey",step.path("stepKey").asText())
                        .put("answerField",answer.path("answerField").asText()).put("answerQuote",quote).put("assessment","PARTIAL")
                        .put("reasoning",explanation).put("improvement",explanation).put("example",explanation);
                finding.putArray("evidenceRefs").add(ref);
                if(!allFields)break;
                }
            }
            if(firstRef==null)throw new IllegalStateException("Harness analysis has no submitted answer");
            crossMajorRefs.add(firstRef);
            var major=majors.addObject().put("majorCode",code).put("preparation",explanation)
                    .put("taskReasoning",explanation).put("nextPractice",explanation);
            major.putArray("evidenceRefs").add(firstRef);
        }
        var comparisons=analysis.putArray("crossMajorAnalysis");
        if("COMPARISON".equals(input.path("reportType").asText())) {
            var comparison=comparisons.addObject().put("observation",explanation)
                    .put("interpretation",explanation).put("nextExperiment",explanation);
            comparison.set("majorCodes",majorCodes);comparison.set("evidenceRefs",crossMajorRefs);
        }
        return proposal;
    }
}
