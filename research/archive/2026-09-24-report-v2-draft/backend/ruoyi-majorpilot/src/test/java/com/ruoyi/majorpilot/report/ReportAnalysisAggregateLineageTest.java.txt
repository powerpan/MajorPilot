    @Test void crossedSumsReconcileBothMarginalsOverallAndExistingNonStarters() throws Exception {
        var facts=context().at("/task/aggregateLineage");
        for(String period:List.of("baseline","comparison"))for(String key:COUNT_KEYS){
            long overall=0;var courses=new java.util.HashMap<String,Long>();var channels=new java.util.HashMap<String,Long>();
            for(var cell:facts.path("crossGroupCounts")){
                long count=cell.path(period).path(key).asLong();overall+=count;String course=cell.path("course_track").asText(),channel=cell.path("signup_source").asText();
                courses.merge(course,count,Long::sum);channels.merge(channel,count,Long::sum);
                if(key.equals("nonStarterCount"))assertEquals(facts.path("periods").path(period).path("nonStartersByCourseAndSource").path(course).path(channel).asLong(),count);
            }
            assertEquals(facts.path("periods").path(period).path("overall").path(key).asLong(),overall);
            for(var entry:courses.entrySet())assertEquals(facts.path("periods").path(period).path("byCourse").path(entry.getKey()).path(key).asLong(),entry.getValue());
            for(var entry:channels.entrySet())assertEquals(facts.path("periods").path(period).path("bySource").path(entry.getKey()).path(key).asLong(),entry.getValue());
        }
    }

    @Test void frozenWindowDenominatorAndGroupingSelectionsControlTheProjection() throws Exception {
        var context=context();var b2=response(context,"b2-metric");
        b2.put("baselineStartWeek",2).put("baselineEndWeek",4).put("comparisonStartWeek",8).put("comparisonEndWeek",10).put("denominator","first_lesson_count");

        response(context,"b6-action").put("targetSegmentKey","CAMPAIGN");
        var parameters=JSON.createObjectNode();for(String key:List.of("baselineStartWeek","baselineEndWeek","comparisonStartWeek","comparisonEndWeek","numerator","denominator"))parameters.set(key,b2.path(key));
        parameters.put("focusTransitionKey","FIRST_LESSON_TO_WEEK_ONE_ACTIVE").put("segmentBy","signup_source");
        var rows=asset("BUSINESS_ANALYTICS","weekly-funnel");String sha=context.at("/task/aggregateLineage/datasetSha256").asText();
        for(var e:context.path("verifiedExecutions"))if(e.at("/value/operationKey").asText().equals("ba-segments"))((ObjectNode)e).set("value",BaStatistics.calculate("ba-segments",rows,parameters,sha));
        enrich(context);var facts=context.at("/task/aggregateLineage");assertEquals(parameters,facts.path("selection"));
        assertEquals(90,facts.at("/periods/baseline/overall/enrolled_count").asInt());
        assertEquals(46,facts.at("/periods/baseline/overall/completed_count").asInt());
        assertEquals(35,facts.at("/periods/comparison/overall/completed_count").asInt());
        assertEquals(8,facts.at("/periods/comparison/startWeek").asInt());assertEquals("signup_source",facts.at("/verifiedSegmentSource/segmentBy").asText());
        for(var cell:facts.path("crossGroupCounts"))for(String period:List.of("baseline","comparison"))for(String key:COUNT_KEYS){
            long expected=0;int start=parameters.path(period+"StartWeek").asInt(),end=parameters.path(period+"EndWeek").asInt();
            for(var row:rows)if(row.path("week").asInt()>=start&&row.path("week").asInt()<=end&&row.path("course_track").equals(cell.path("course_track"))&&row.path("signup_source").equals(cell.path("signup_source")))
                expected+=key.equals("nonStarterCount")?row.path("enrolled_count").asLong()-row.path("first_lesson_count").asLong():row.path(key).asLong();
            assertEquals(expected,cell.path(period).path(key).asLong(),period+" "+key);
        }
    }
    @Test void unknownSchemaMixedVersionsAndMismatchedResultsCannotProduceTrustedFacts() throws Exception {
        for(String fault:List.of("context-schema","task-schema","dictionary-schema","engine-schema","dataset-hash","selected-window","execution-count","raw-count","valid-raw-count","duplicate-material")){
            var context=context();var manifest=(ObjectNode)asset("BUSINESS_ANALYTICS","manifest").deepCopy();
            switch(fault){
                case "context-schema"->context.put("schemaVersion","unknown-context");
                case "task-schema"->manifest.put("schemaVersion","unknown-task");
                case "dictionary-schema"->((ObjectNode)material(context,"data-dictionary")).put("schemaVersion","unknown-dictionary");
                case "engine-schema"->{for(var op:manifest.path("operations"))if(op.path("operationKey").asText().equals("ba-segments"))((ObjectNode)op).put("engineVersion","unknown-engine");}
                case "dataset-hash"->{for(var a:manifest.path("assets"))if(a.path("assetKey").asText().equals("weekly-funnel"))((ObjectNode)a).put("sha256","0".repeat(64));}
                case "selected-window"->response(context,"b2-metric").put("comparisonStartWeek",8);
                case "execution-count"->{for(var e:context.path("verifiedExecutions"))if(e.at("/value/operationKey").asText().equals("ba-segments"))((ObjectNode)e.at("/value/segments/0/comparison/stageCounts")).put("completed_count",26);}
                case "raw-count"->((ObjectNode)material(context,"weekly-funnel").get(0)).put("enrolled_count",-1);
                case "valid-raw-count"->{var row=(ObjectNode)material(context,"weekly-funnel").get(0);row.put("enrolled_count",row.path("enrolled_count").asLong()+1);}
                default->((com.fasterxml.jackson.databind.node.ArrayNode)context.at("/task/materials")).add(context.at("/task/materials/0").deepCopy());
            }
            // Repeated or foreign material sources may not silently replace the pinned inputs.
            if(fault.equals("duplicate-material")){
                var materials=(com.fasterxml.jackson.databind.node.ArrayNode)context.at("/task/materials");
                for(var item:materials)if(item.path("assetKey").asText().equals("weekly-funnel")){materials.add(item.deepCopy());break;}
            }
            assertEquals("REPORT_SOURCE_INVALID",assertThrows(ApiException.class,()->ReportAnalysisContext.enrichAggregateLineage(JSON,context,manifest,ReportAnalysisContext.RUBRIC),fault).code);
        }
    }
