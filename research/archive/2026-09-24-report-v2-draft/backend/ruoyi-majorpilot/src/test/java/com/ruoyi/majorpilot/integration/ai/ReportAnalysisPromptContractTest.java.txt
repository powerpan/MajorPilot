package com.ruoyi.majorpilot.integration.ai;

import static org.junit.jupiter.api.Assertions.*;
import org.junit.jupiter.api.Test;

/** Offline prompt-policy contract, not evidence that a model obeys it or writes correct analysis. */
class ReportAnalysisPromptContractTest {
    private String prompt()throws Exception{var field=ReportHttpGateway.class.getDeclaredField("ANALYSIS_PROMPT");field.setAccessible(true);return (String)field.get(null);}
    private String reviewPrompt()throws Exception{var field=ReportHttpGateway.class.getDeclaredField("REVIEW_PROMPT");field.setAccessible(true);return (String)field.get(null);}
    @Test void findingsUseOnlyTheOpenAnswerWhitelistWhileReadingStructuredContext()throws Exception{String p=prompt();for(String rule:new String[]{"finding.answerField 只能选择对应 steps[].answers[].answerField","开放题白名单","不得从 submittedResponse 的任意其他键名自行创建 finding","结构化选择、数值与数组仍须完整阅读","不可忽略，也不另造开放题字段"})assertTrue(p.contains(rule),rule);}
    @Test void majorAnalysesBindAtLeastOneExistingOriginalAnswerFromTheSameMajor()throws Exception{String p=prompt();for(String rule:new String[]{"每项 evidenceRefs 必须至少包含一条该专业原始作答引用","type=STUDENT_ANALYSIS 的 evidenceId","sourceType=STEP_RESPONSE 的 sourceRef","不能仅引用 VERIFIED_EXECUTION 等结果记录","与该项分析相关","不得凭空补造引用","作答引用只证明学生写过，不证明其观点正确"})assertTrue(p.contains(rule),rule);}
    @Test void checksAllSubmittedFieldsBeforeCallingInformationMissing()throws Exception{String p=prompt();assertTrue(p.contains("submittedResponse"));assertTrue(p.contains("不能把已经填写或提供的信息说成缺失"));assertTrue(p.contains("每条批评先检查相同答案其他字段是否已补足条件"));}
    @Test void examplesHaveTheSameEvidenceStandardAndNumbersAreNotReadyMadeGuarantees()throws Exception{String p=prompt();assertTrue(p.contains("example使用同一证据标准与一致判断"));assertTrue(p.contains("样本量充分性"));assertTrue(p.contains("效果/成本/负荷保证"));assertTrue(p.contains("假设示例、需基线/效应/不确定性与资源论证后确定"));assertTrue(p.contains("不能当作既有事实或现成阈值"));}
    @Test void distinguishesMechanismFromInterventionAndRequiresConfoundingAndStatisticalCautions()throws Exception{String p=prompt();assertTrue(p.contains("干预未奏效不能直接证伪机制假设"));assertTrue(p.contains("时间与构成混杂"));assertTrue(p.contains("没有相应检验不使用“统计显著/未显著”等表述"));}
    @Test void safeguardsAreGenericRatherThanAnswersToSpecificPaidFixtures()throws Exception{String p=prompt();for(String fixtureDetail:new String[]{"ba-good","ba-concept-errors","114人","149.1","30-50人","APPLIED的报名"})assertFalse(p.contains(fixtureDetail),fixtureDetail);assertTrue(p.contains("允许不同于示例的合理解法"));assertTrue(p.contains("不能断言学生故意乱填"));}
    @Test void evaluatesTheLocalQuestionWithoutReplacingALegitimateOriginalStrategy()throws Exception{String p=prompt();assertTrue(p.contains("当前步骤题干、字段要求和本题评价准则"));assertTrue(p.contains("不可用后续步骤才要求的完整实验条件"));assertTrue(p.contains("不是本题必答项或已犯错误"));assertTrue(p.contains("保留其合法重点和策略"));assertTrue(p.contains("不要把你的另一种偏好偷换成标准答案"));}
    @Test void usesExistingMaterialsAndRechecksAnyNewArithmetic()throws Exception{String p=prompt();assertTrue(p.contains("先读取、计算"));assertTrue(p.contains("不把已有信息列为仍需采集"));assertTrue(p.contains("所有新增算式、比例、百分点、基线与目标关系须逐项复核"));assertTrue(p.contains("无法可靠计算就不新增具体数值"));assertTrue(p.contains("学生已填的目标应称为其既有目标"));}
    @Test void initialAndReviewAssessTheCurrentQuestionRatherThanDemandingPerfection()throws Exception{
        for(String p:new String[]{prompt(),reviewPrompt()})for(String rule:new String[]{"SUPPORTED 不等于完美或没有改进空间","当前题干和本题评价准则的必要要求已满足","PARTIAL 必须在 reasoning 指出当前题干或本题准则的哪一项必要条件尚未满足","NEEDS_REVISION 必须对应本题的明确错误或矛盾","只有后续步骤或进阶练习的建议，不得将 SUPPORTED 降为 PARTIAL 或 NEEDS_REVISION","若 reasoning 承认本题要求已满足，assessment 必须与此一致","后续建议明确标为可选，不得重新描述为本题缺口"})assertTrue(p.contains(rule),rule);
    }
    @Test void reviewInventoriesGivenAndDerivableFactsBeforeCallingAnythingMissing()throws Exception{String p=reviewPrompt();for(String rule:new String[]{"先盘点已知信息，再作评价","服务端提供的派生事实及其来源链","直接提供","可按已给口径推导","需要额外假设","确未提供","不能被写成未知、缺失或需要重新收集","不能把汇总边际分布擅自展开为未提供的交叉分布"})assertTrue(p.contains(rule),rule);}
    @Test void reviewAcceptsTraceableAggregationRatherThanRequiringTheSameDisplayStep()throws Exception{String p=reviewPrompt();for(String rule:new String[]{"合法聚合应按来源链追溯","不硬性要求结论与证据出现在同一步","互斥且覆盖目标总体的分组计数可以加总","分子总和除以分母总和","不能直接平均组内比率","只属于表达优化，不能写成事实错误或强制换引","重叠、缺组、范围或口径不一致"})assertTrue(p.contains(rule),rule);}
    @Test void reviewExamplesCannotReintroduceMissingFactsOrUnsupportedCausalControls()throws Exception{String p=reviewPrompt();for(String rule:new String[]{"每个示例必须重新通过同一已知信息盘点","不能在正文承认已有事实，却在示例中将其重新写成未知","跨课程或自然群体","不等于随机分配的因果对照","即使同期也不能自动排除","不把非随机跨课程差异称为干预净效应","不覆盖学生已填目标、已给数据或原合法方案"})assertTrue(p.contains(rule),rule);}
    @Test void studentNarrativeAvoidsInternalKeysWithoutChangingMachineBindingsOrQuotes()throws Exception{String p=reviewPrompt();for(String rule:new String[]{"面向学生的分析正文使用自然语言的题目名称和指标名称","不展示内部字段键名、步骤键名、枚举代码、JSON路径或来源ID","合同要求的机器字段及归属值仍按原格式返回","evidenceRefs仍使用允许引用","answerQuote仍须逐字保留原文"})assertTrue(p.contains(rule),rule);}
    @Test void initialNarrativeNeverEchoesReferenceIdsWhileMachineRefsAndExactQuotesRemainIntact()throws Exception{String p=prompt();for(String rule:new String[]{"面向学生的新写正文不展示内部证据ID、来源ID、sourceRef","不要在 reasoning、summary、improvement、example 等正文中用括号回显来源编号","用自然语言的题目名称和指标名称解释依据","合同要求的机器字段及归属值仍按原格式返回","evidenceRefs仍使用输入允许的真实引用","answerQuote仍须逐字保留学生原文","即使原话包含键名或编号，也不能改写或删去"})assertTrue(p.contains(rule),rule);}
    @Test void reviewFactRulesDoNotEmbedParticularCasesValuesOrExpectedAnswers()throws Exception{String p=reviewPrompt();for(String fixture:new String[]{"ba-good","ba-concept-errors","weekly-funnel","b4-segment","100/180","58/179","23.15","114人","149.1"})assertFalse(p.contains(fixture),fixture);}
    @Test void reviewDoesNotCallBundledIndependentChangesASingleFactorExperiment()throws Exception{String p=reviewPrompt();for(String rule:new String[]{"声称单变量试验时只能改变一个独立因素","即使都属于同一类别或群体","不能称为单变量或把结果归因给其中一个因素","分轮分别改变因素","只评估组合方案的整体效果"})assertTrue(p.contains(rule),rule);}
}
