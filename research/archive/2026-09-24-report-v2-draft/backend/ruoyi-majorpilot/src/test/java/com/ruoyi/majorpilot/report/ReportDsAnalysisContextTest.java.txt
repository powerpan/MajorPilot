    @Test void mismatchedSourceChoicesAndDenominatorsFailClosed() throws Exception {
        for(String fault:List.of("dataset","cleaning","axes","profile","stratifier","denominator","missing-execution","dictionary-duplicate")){
            var context=context();
            switch(fault){
                case "dataset"->execution(context,"ds-profile").put("datasetSha256","f".repeat(64));
                case "cleaning"->response(context,"d2-quality").put("missingPolicy","MEDIAN_NUMERIC");
                case "axes"->response(context,"d4-relationship").put("xField","avg_session_minutes");
                case "profile"->response(context,"d3-profile").put("distributionField","days_to_first_lesson");
                case "stratifier"->response(context,"d5-stratify").put("stratifyBy","cohort_period");
                case "denominator"->((ObjectNode)execution(context,"ds-profile").path("profile")).put("validCount",123);
                case "missing-execution"->((ArrayNode)context.path("verifiedExecutions")).remove(3);
                default->{var fields=(ArrayNode)context.at("/task/materials/0/content/fields");fields.add(fields.get(0).deepCopy());}
            }
            assertEquals("REPORT_SOURCE_INVALID",assertThrows(ApiException.class,()->enrich(context),fault).code);
            assertFalse(context.path("task").has("analysisFrame"),"Partial frame must never escape a failed check");
        }
    }

    @Test void sameResultMayHaveMultipleRealReferencesButConflictingResultsAreRejected() throws Exception {
        var context=context();var executions=(ArrayNode)context.path("verifiedExecutions");var copy=executions.get(2).deepCopy();((ObjectNode)copy).put("evidenceId",id(99999));executions.add(copy);
        enrich(context);assertEquals(2,analysis(context.at("/task/analysisFrame"),"ds-relationship").path("evidenceRefs").size());
        ((ObjectNode)copy.at("/value/relationship/scatter")).put("r",.8);
        assertEquals("REPORT_SOURCE_INVALID",assertThrows(ApiException.class,()->enrich(context)).code);
    }

    private static ObjectNode context() throws Exception {var context=(ObjectNode)ReportAnalysisLiveFixtures.input("ds-causal-error").sources().getFirst().analysisContext().deepCopy();((ObjectNode)context.path("task")).remove("analysisFrame");return context;}
    private static void enrich(ObjectNode context){ReportDsAnalysisContext.enrich(JSON,context,asset("DATA_SCIENCE","manifest"));}
    private static JsonNode analysis(JsonNode frame,String operation){for(var analysis:frame.path("analyses"))if(operation.equals(analysis.path("operationKey").asText()))return analysis;throw new AssertionError(operation);}
    private static ObjectNode response(JsonNode context,String step){for(var entry:context.path("steps"))if(step.equals(entry.path("stepKey").asText()))return (ObjectNode)entry.path("submittedResponse");throw new AssertionError(step);}
    private static ObjectNode execution(JsonNode context,String operation){for(var evidence:context.path("verifiedExecutions"))if(operation.equals(evidence.at("/value/operationKey").asText()))return (ObjectNode)evidence.path("value");throw new AssertionError(operation);}
    private static void recalculate(ObjectNode context){
        var rows=asset("DATA_SCIENCE","learner-activity");var datasetSha=execution(context,"ds-quality-clean").path("datasetSha256").asText();
        for(var evidence:context.path("verifiedExecutions")){
            String operation=evidence.at("/value/operationKey").asText();var parameters=JSON.createObjectNode();for(String key:DsSemantics.CLEAN)parameters.set(key,response(context,"d2-quality").path(key));
            if(operation.equals("ds-profile"))parameters.set("distributionField",response(context,"d3-profile").path("distributionField"));
            if(List.of("ds-relationship","ds-stratify").contains(operation))for(String key:DsSemantics.RELATION)if(response(context,"d4-relationship").has(key))parameters.set(key,response(context,"d4-relationship").path(key));
            if(operation.equals("ds-stratify"))parameters.set("stratifyBy",response(context,"d5-stratify").path("stratifyBy"));
            ((ObjectNode)evidence).set("value",DsStatistics.calculate(operation,rows,parameters,datasetSha));
        }
    }
}
