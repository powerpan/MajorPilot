    /** Deliberately labelled no-network test output; never product prose or a model-quality assertion. */
    JsonNode syntheticAnalysis(SafeModelInput safe){
        // This no-network stub supplies an already-decoded internal proposal. Bound provider inputs
        // carry E aliases, so their raw references are read only from the local, nonserialized catalog.
        boolean bound=safe.bindingCatalogContext()!=null;
        JsonNode input=bound?safe.bindingCatalogContext().analysisContext():safe.content().has("originalInput")?safe.content().path("originalInput"):safe.content();
        String explanation="Synthetic transactional test explanation grounded only in the supplied submission; not a real model assessment.";
        var root=mapper.createObjectNode().put("schemaVersion","report-analysis-proposal-v2").put("reportType",input.path("reportType").asText());var analysis=root.putObject("analysis");analysis.put("summary",explanation);var majors=analysis.putArray("majorAnalyses");var findings=analysis.putArray("findings");var refs=mapper.createArrayNode();var codes=mapper.createArrayNode();
        for(var source:input.path("sources")){String code=source.path("majorCode").asText();codes.add(code);String firstRef=null;
            for(var step:source.at("/context/steps"))for(var answer:step.path("answers")){String ref=answer.path(answer.has("evidenceId")?"evidenceId":"sourceRef").asText();if(firstRef==null)firstRef=ref;
                String quote=answer.path("value").asText();if(bound&&quote.codePointCount(0,quote.length())>1200)quote=quote.substring(0,quote.offsetByCodePoints(0,1200));
                var finding=findings.addObject().put("majorCode",code).put("stepKey",step.path("stepKey").asText()).put("answerField",answer.path("answerField").asText()).put("answerQuote",quote).put("assessment","PARTIAL").put("reasoning",explanation).put("improvement",explanation).put("example",explanation);finding.putArray("evidenceRefs").add(ref);
                if(!bound)break; // Keep historical v2 checkpoint shape; bound wire covers every slot.
            }
            assertNotNull(firstRef);refs.add(firstRef);var major=majors.addObject().put("majorCode",code).put("preparation",explanation).put("taskReasoning",explanation).put("nextPractice",explanation);major.putArray("evidenceRefs").add(firstRef);
        }
        var comparisons=analysis.putArray("crossMajorAnalysis");if(input.path("reportType").asText().equals("COMPARISON")){var comparison=comparisons.addObject().put("observation",explanation).put("interpretation",explanation).put("nextExperiment",explanation);comparison.set("majorCodes",codes);comparison.set("evidenceRefs",refs);}return root;
    }
