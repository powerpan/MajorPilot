class PersonalExportWriterTest {
    @Test void v2AnalysisExportsQuotedAnswersAndExamplesAsEscapedText()throws Exception{
        var body=(ObjectNode)golden();var analysis=body.putObject("analysis");analysis.put("status","COMPLETE").put("summary","本次存在指标混淆");analysis.putArray("majorAnalyses").addObject().put("majorCode","BUSINESS_ANALYTICS").put("preparation","当前概念").put("taskReasoning","推理解释").put("nextPractice","下一练习");analysis.putArray("findings").addObject().put("majorCode","BUSINESS_ANALYTICS").put("stepTitle","行动计划").put("assessment","NEEDS_REVISION").put("answerQuote","<script>用户原话</script>").put("reasoning","具体原因").put("improvement","具体改进").put("example","改写示例");analysis.putArray("crossMajorAnalysis");
        String html=PersonalExportWriter.html(body);for(String required:List.of("你的作答分析","当前概念","推理解释","下一练习","具体原因","具体改进","改写示例","&lt;script&gt;用户原话&lt;/script&gt;"))assertTrue(html.contains(required));assertFalse(html.contains("<script>"));assertFalse(html.contains("evidenceRefs"));
        analysis.put("status","UNAVAILABLE");String fallback=PersonalExportWriter.html(body);assertTrue(fallback.contains("尚未完成"));assertFalse(fallback.contains("具体原因"));
    }
    @Test void adoptedPreviewExportPreservesOriginalModeAndOnlySafePersonalProvenance()throws Exception{
        var run=new LinkedHashMap<String,Object>();run.put("id",9L);run.put("public_id","00000000000000000000000009");run.put("run_mode","REVIEW_PREVIEW");run.put("content_kind","MAJOR");run.put("task_public_id","00000000000000000000000010");run.put("profile_public_id","00000000000000000000000011");run.put("manifest_json","{\"title\":\"SYNTHETIC\"}");
        when(db.queryForList(startsWith("SELECT r.id,r.public_id,r.content_kind"),eq(7L))).thenReturn(List.of(run));
        var proof=new LinkedHashMap<String,Object>();proof.put("public_id","00000000000000000000000012");proof.put("original_run_mode","REVIEW_PREVIEW");proof.put("authority_code","PROJECT_OWNER_REQUEST");proof.put("authorized_at",Instant.parse("2026-09-21T00:00:00Z"));proof.put("authorized_by_operator_id",777L);proof.put("reason_text","PRIVATE_OPERATOR_REASON");proof.put("authorization_reference","PRIVATE_AUTHORIZATION_REFERENCE");proof.put("closed_input_sha256","PRIVATE_HASH");
        when(db.queryForList(startsWith("SELECT a.* FROM mp_run_product_adoption"),eq(7L),eq(9L))).thenReturn(List.of(proof));
        var files=unzip(writer.build(7,"FULL_PERSONAL_EXPORT",Instant.now()));var data=mapper.readTree(files.get("personal-data.json"));var exported=data.path("runs").get(0);
        assertEquals("REVIEW_PREVIEW",exported.path("runMode").asText());var visible=exported.path("productAuthorization");var keys=new HashSet<String>();visible.fieldNames().forEachRemaining(keys::add);
        assertEquals(Set.of("authorizationId","experienceRunId","taskVersionId","decisionProfileId","originalRunMode","authorityCode","authorizedAt"),keys);
        assertEquals("PROJECT_OWNER_REQUEST",visible.path("authorityCode").asText());
        for(String forbidden:List.of("PRIVATE_","authorized_by_operator_id","reason_text","authorization_reference","closed_input_sha256"))assertFalse(data.toString().contains(forbidden),forbidden);
        var reportsOnly=unzip(writer.build(7,"REPORTS_ONLY",Instant.now()));assertFalse(reportsOnly.containsKey("personal-data.json"));
    }

    final ObjectMapper mapper=new ObjectMapper().findAndRegisterModules();final JdbcTemplate db=mock(JdbcTemplate.class);PersonalExportWriter writer;
    @BeforeEach void setup(){writer=new PersonalExportWriter(db,mapper);TransactionSynchronizationManager.setActualTransactionActive(true);when(db.queryForList(startsWith("SELECT public_id,email_display,status"),eq(7L))).thenReturn(List.of(new LinkedHashMap<>(Map.of("public_id","00000000000000000000000007","email_display","export@example.invalid","status","ACTIVE"))));}
    @AfterEach void clear(){TransactionSynchronizationManager.setActualTransactionActive(false);}
    static Map<String,byte[]> unzip(byte[] bytes)throws Exception{var out=new LinkedHashMap<String,byte[]>();try(var zip=new ZipInputStream(new ByteArrayInputStream(bytes),StandardCharsets.UTF_8)){for(var entry=zip.getNextEntry();entry!=null;entry=zip.getNextEntry())assertNull(out.put(entry.getName(),zip.readAllBytes()));}return out;}
    JsonNode golden()throws Exception{return mapper.readTree(getClass().getResourceAsStream("/report-goldens/ai-fallback.json"));}
    void report(JsonNode body,String hash){when(db.queryForList(startsWith("SELECT p.public_id AS report_public_id"),eq(7L))).thenReturn(List.of(Map.of("report_public_id",body.path("reportId").asText(),"public_id",body.path("reportVersionId").asText(),"version_no",body.path("versionNo").longValue(),"status","SUPERSEDED","structured_report_json",body.toString(),"report_body_sha256",hash)));when(db.queryForObject(startsWith("SELECT COUNT(*) FROM mp_experience_run"),eq(Integer.class),any(),eq(7L),any())).thenReturn(1);}
    @Test void requiresCallerTransactionBeforeAnyQueries(){TransactionSynchronizationManager.setActualTransactionActive(false);assertEquals("PERSONAL_EXPORT_SNAPSHOT_REQUIRED",assertThrows(IllegalStateException.class,()->writer.build(7,"FULL_PERSONAL_EXPORT",Instant.now())).getMessage());verifyNoInteractions(db);}
    @Test void reportOnlyHasManifestAndNoPersonalDataAndKeepsPublishedHistoricalVersion()throws Exception{var body=golden();report(body,CanonicalJson.hash(mapper,body));var files=unzip(writer.build(7,"REPORTS_ONLY",Instant.parse("2026-09-21T00:00:00Z")));assertEquals(Set.of("manifest.json","reports/"+body.path("reportVersionId").asText()+".html"),files.keySet());var manifest=mapper.readTree(files.get("manifest.json"));assertEquals("personal-export-v1",manifest.path("schemaVersion").asText());assertEquals("SUPERSEDED",manifest.path("files").get(0).path("publicationStatus").asText());assertFalse(new String(files.get("manifest.json"),StandardCharsets.UTF_8).contains("export@example.invalid"));verify(db,never()).queryForList(startsWith("SELECT r.id,r.public_id,r.content_kind"),anyLong());}
    @Test void fullExportHasClosedTopLevelSchemaAndReadableAccountOnly()throws Exception{var files=unzip(writer.build(7,"FULL_PERSONAL_EXPORT",Instant.now()));var data=mapper.readTree(files.get("personal-data.json"));var keys=new HashSet<String>();data.fieldNames().forEachRemaining(keys::add);assertEquals(Set.of("schemaVersion","generatedAt","account","consents","claimedIntakes","profiles","questionnaire","recommendations","runs","feedback"),keys);assertEquals("export@example.invalid",data.path("account").path("email").asText());assertFalse(data.toString().contains("password"));assertFalse(data.toString().contains("session"));assertFalse(data.path("account").has("id"));assertEquals("personal-data-v1",data.path("schemaVersion").asText());}
    @Test void pendingVerificationCanExportBothScopesWithoutOpeningTaskAccess()throws Exception{when(db.queryForList(startsWith("SELECT public_id,email_display,status"),eq(7L))).thenReturn(List.of(Map.of("public_id","00000000000000000000000007","email_display","pending@example.invalid","status","PENDING_VERIFICATION")));for(String scope:List.of("REPORTS_ONLY","FULL_PERSONAL_EXPORT")){var files=unzip(writer.build(7,scope,Instant.now()));assertTrue(files.containsKey("manifest.json"));if(scope.equals("FULL_PERSONAL_EXPORT"))assertEquals("PENDING_VERIFICATION",mapper.readTree(files.get("personal-data.json")).path("account").path("status").asText());}}
    @Test void lockedDeletingDeletedAndMissingAccountsCannotBeExported(){for(String status:List.of("LOCKED","DELETION_PENDING","DELETED")){when(db.queryForList(startsWith("SELECT public_id,email_display,status"),eq(7L))).thenReturn(List.of(Map.of("public_id","00000000000000000000000007","status",status)));assertEquals("ACCOUNT_UNAVAILABLE",assertThrows(ApiException.class,()->writer.build(7,"FULL_PERSONAL_EXPORT",Instant.now())).code);}when(db.queryForList(startsWith("SELECT public_id,email_display,status"),eq(7L))).thenReturn(List.of());assertEquals("ACCOUNT_UNAVAILABLE",assertThrows(ApiException.class,()->writer.build(7,"REPORTS_ONLY",Instant.now())).code);}
    @Test void invalidPublishedBodyOrHashFailsWholePackageRatherThanSkipping()throws Exception{var body=golden();report(body,"0".repeat(64));assertEquals("PERSONAL_EXPORT_SOURCE_INVALID",assertThrows(ApiException.class,()->writer.build(7,"REPORTS_ONLY",Instant.now())).code);((ObjectNode)body).put("privatePrompt","must never leak");report(body,CanonicalJson.hash(mapper,body));assertEquals("PERSONAL_EXPORT_SOURCE_INVALID",assertThrows(ApiException.class,()->writer.build(7,"REPORTS_ONLY",Instant.now())).code);}
    @Test void foreignReportSourceRejectsEvenAValidPublishedBody()throws Exception{var body=golden();report(body,CanonicalJson.hash(mapper,body));when(db.queryForObject(startsWith("SELECT COUNT(*) FROM mp_experience_run"),eq(Integer.class),any(),eq(7L),any())).thenReturn(0);assertEquals("PERSONAL_EXPORT_SOURCE_INVALID",assertThrows(ApiException.class,()->writer.build(7,"REPORTS_ONLY",Instant.now())).code);}
    @Test void htmlDoesNotInterpretMarkupOrExternalLinks()throws Exception{var body=golden();String attack="</li><script>alert(1)</script><img src=\"https://evil.invalid/x\" onerror='bad'> & 用户原文";((ObjectNode)body.path("sections").get(0).path("claims").get(0)).put("text",attack);String html=PersonalExportWriter.html(body);assertFalse(html.contains("<script"));assertFalse(html.contains("<img"));assertFalse(html.contains("href="));assertTrue(html.contains("&lt;script&gt;alert(1)&lt;/script&gt;"));assertTrue(html.contains("&quot;https://evil.invalid/x&quot;"));assertTrue(html.contains("&#39;bad&#39;"));assertTrue(html.contains("&amp; 用户原文"));assertTrue(html.contains("default-src 'none'"));}
    @Test void zipRejectsPathsDuplicatesAndExpandedLimitEvenWhenHighlyCompressible(){var zip=new PersonalExportWriter.Archive(2048);assertThrows(ApiException.class,()->zip.add("../outside",new byte[1]));var one=new PersonalExportWriter.Archive(2048);one.add("manifest.json",new byte[100]);assertThrows(ApiException.class,()->one.add("manifest.json",new byte[1]));var expanded=new PersonalExportWriter.Archive(2048);expanded.add("manifest.json",new byte[1500]);assertEquals("PERSONAL_EXPORT_TOO_LARGE",assertThrows(ApiException.class,()->expanded.add("personal-data.json",new byte[549])).code);}
    @Test void compressedArchiveOverheadAlsoCannotBypassHardLimit(){var archive=new PersonalExportWriter.Archive(100);assertEquals("PERSONAL_EXPORT_TOO_LARGE",assertThrows(ApiException.class,()->{archive.add("manifest.json",new byte[1]);archive.finish();}).code);}
}
