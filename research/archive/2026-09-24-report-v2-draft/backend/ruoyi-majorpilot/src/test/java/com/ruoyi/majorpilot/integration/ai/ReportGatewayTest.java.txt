package com.ruoyi.majorpilot.integration.ai;
import com.fasterxml.jackson.databind.*;
import com.ruoyi.majorpilot.report.core.*;
import static com.ruoyi.majorpilot.report.core.ReportTestFixtures.*;
import static org.junit.jupiter.api.Assertions.*;
import java.nio.charset.StandardCharsets;
import java.util.*;
import java.util.concurrent.atomic.AtomicInteger;
import org.junit.jupiter.api.Test;
import org.springframework.mock.env.MockEnvironment;

class ReportGatewayTest {
    final ReportCore core=new ReportCore(JSON);
    ReportGateway.Attempt attempt(){var c=core.compile(experience("AI",true));return new ReportGateway.Attempt("synthetic-invocation",ReportGateway.Stage.INITIAL,core.modelInput(c),List.of(),45000);}
    byte[] envelope(JsonNode proposal)throws Exception{return JSON.writeValueAsBytes(Map.of("choices",List.of(Map.of("finish_reason","stop","message",Map.of("content",proposal.toString())))));}
    @Test void bothProvidersUseOneAllowlistedHttpAndOnlySafeCatalog()throws Exception{for(String provider:List.of("BAILIAN","DEEPSEEK")){var calls=new AtomicInteger();var input=attempt();var gateway=new ReportHttpGateway(JSON,core,provider,"synthetic-key",provider.equals("BAILIAN")?BailianAiGateway.DEFAULT_BASE_URL:ReportHttpGateway.DEEPSEEK_BASE_URL,provider.equals("BAILIAN")?"qwen3.7-plus":"deepseek-flash",(request,timeout)->{calls.incrementAndGet();assertEquals("https",request.uri().getScheme());assertEquals(45000,timeout);try{return new BailianAiGateway.TransportResponse(200,envelope(input.input().content().path("completeValidProposal")));}catch(Exception e){throw new AssertionError(e);}});var result=gateway.attempt(input);assertEquals(ReportGateway.Status.PROPOSAL_RECEIVED,result.status());assertEquals(1,calls.get());assertEquals(1,result.telemetry().httpAttempts());assertNull(result.telemetry().inputTokens());assertNull(result.telemetry().estimatedCost());assertFalse(input.input().content().toString().contains("PRIVATE_SENTINEL"));}}
    @Test void badJson429AndTimeoutNeverPerformHiddenRetryOrLeakBody(){for(String kind:List.of("BAD_JSON","429","TIMEOUT")){var calls=new AtomicInteger();var g=new ReportHttpGateway(JSON,core,"DEEPSEEK","synthetic-key",ReportHttpGateway.DEEPSEEK_BASE_URL,"deepseek-flash",(req,timeout)->{calls.incrementAndGet();if(kind.equals("TIMEOUT"))throw new AiFailure(AiFailure.Code.TIMEOUT);return new BailianAiGateway.TransportResponse(kind.equals("429")?429:200,"PRIVATE_SENTINEL_PROVIDER_BODY".getBytes(StandardCharsets.UTF_8));});var output=g.attempt(attempt());assertEquals(1,calls.get());assertEquals(1,output.telemetry().httpAttempts());assertNull(output.proposal());assertFalse(output.toString().contains("PRIVATE_SENTINEL"));}}
    @Test void disallowedUrlsAndDefaultDisabledDoNotCallAnything(){for(String url:List.of("http://api.deepseek.com","https://api.deepseek.com.evil.test","https://u:p@api.deepseek.com","https://api.deepseek.com:443","https://api.deepseek.com/v1?key=x","https://localhost"))assertThrows(AiFailure.class,()->ReportHttpGateway.deepseekEndpoint(url));var configured=new ReportConfiguredGateway(JSON,core,new MockEnvironment().withProperty("MAJORPILOT_AI_MODE","bailian").withProperty("BAILIAN_API_KEY","synthetic-key"));assertFalse(configured.available());assertThrows(AiFailure.class,()->configured.attempt(attempt()));}
    @Test void repairIsExplicitSingleAttemptAndBudgetCannotReset(){var calls=new AtomicInteger();var g=new ReportHttpGateway(JSON,core,"BAILIAN","synthetic-key",BailianAiGateway.DEFAULT_BASE_URL,"qwen3.7-plus",(req,timeout)->{calls.incrementAndGet();assertEquals(12,timeout);return new BailianAiGateway.TransportResponse(500,new byte[0]);});var original=attempt();var repair=new ReportGateway.Attempt(original.invocationId(),ReportGateway.Stage.REPAIR,original.input(),List.of(new ReportModels.Violation("SECTION_ORDER_INVALID","/sections")),12);assertEquals("UPSTREAM_UNAVAILABLE",g.attempt(repair).safeErrorCode());assertEquals(1,calls.get());var invalid=new ReportGateway.Attempt(original.invocationId(),ReportGateway.Stage.REPAIR,original.input(),List.of(),ReportModelSettings.MAX_TOTAL_BUDGET_MS+1);assertEquals(0,g.attempt(invalid).telemetry().httpAttempts());assertEquals(1,calls.get());}
    @Test void routingIdentityIsReadOnlyAndCannotExposeCredentialsOrInventDisabledIdentity(){
        for(String provider:List.of("BAILIAN","DEEPSEEK")){
            String model=provider.equals("BAILIAN")?ReportHttpGateway.BAILIAN_MODEL:ReportHttpGateway.DEEPSEEK_MODEL;
            var direct=new ReportHttpGateway(JSON,core,provider,"SECRET_SENTINEL",provider.equals("BAILIAN")?BailianAiGateway.DEFAULT_BASE_URL:ReportHttpGateway.DEEPSEEK_BASE_URL,model,(request,timeout)->{throw new AssertionError("Identity inspection must not call HTTP");});
            assertEquals(new ReportGateway.Identity(provider,model),direct.identity());assertTrue(direct.identity().known());assertFalse(direct.identity().toString().contains("SECRET_SENTINEL"));
            var env=new MockEnvironment().withProperty("MP_REPORT_AI_ENABLED","true").withProperty("MP_REPORT_AI_PROVIDER",provider).withProperty(provider+"_API_KEY","SECRET_SENTINEL").withProperty("MP_REPORT_AI_MODEL",model);
            assertEquals(direct.identity(),new ReportConfiguredGateway(JSON,core,env).identity());
        }
        assertNull(new ReportConfiguredGateway(JSON,core,new MockEnvironment()).identity());
        for(String unknown:Arrays.asList(null,"","UNKNOWN","pending","UNAVAILABLE","DISABLED"," with-spaces ")){assertFalse(new ReportGateway.Identity(unknown,"valid-model").known());assertFalse(new ReportGateway.Identity("BAILIAN",unknown).known());}
    }
}
