    JsonNode body(HttpRequest request) {
        var bytes = new ByteArrayOutputStream();
        var ready = new CompletableFuture<JsonNode>();
        request.bodyPublisher().orElseThrow().subscribe(new Flow.Subscriber<ByteBuffer>() {
            public void onSubscribe(Flow.Subscription value) { value.request(Long.MAX_VALUE); }
            public void onNext(ByteBuffer buffer) { byte[] chunk = new byte[buffer.remaining()]; buffer.get(chunk); bytes.writeBytes(chunk); }
            public void onError(Throwable error) { ready.completeExceptionally(error); }
            public void onComplete() { try { ready.complete(JSON.readTree(bytes.toByteArray())); } catch (Exception error) { ready.completeExceptionally(error); } }
        });
        try { return ready.get(2, TimeUnit.SECONDS); } catch (Exception error) { throw new AssertionError(error); }
    }
    byte[] envelope(ReportGateway.Attempt attempt, String reasoning) {
        try {
            return JSON.writeValueAsBytes(Map.of("choices", List.of(Map.of("finish_reason", "stop", "message", Map.of(
                    "content", attempt.input().content().path("completeValidProposal").toString(), "reasoning_content", reasoning))),
                    "usage", Map.of("prompt_tokens", 5000, "completion_tokens", 6200, "completion_tokens_details", Map.of("reasoning_tokens", 5000))));
        } catch (Exception error) { throw new AssertionError(error); }
    }

    @Test void bothProvidersKeepLegacyWireFormatAnd45SecondJobBudget() {
        for (String provider : List.of("BAILIAN", "DEEPSEEK")) {
            var input = legacyAttempt();
            var gateway = new ReportHttpGateway(JSON, core, provider, "synthetic-key",
                    provider.equals("BAILIAN") ? BailianAiGateway.DEFAULT_BASE_URL : ReportHttpGateway.DEEPSEEK_BASE_URL,
                    provider.equals("BAILIAN") ? ReportHttpGateway.BAILIAN_MODEL : ReportHttpGateway.DEEPSEEK_MODEL,
                    (request, timeout) -> {
                        assertEquals(45_000, timeout);
                        assertEquals(timeout, request.timeout().orElseThrow().toMillis());
                        var sent = body(request);
                        assertFalse(sent.path("stream").asBoolean(), "Historical v1 remains non-streaming");
                        assertFalse(sent.has("stream_options"));
                        assertTrue(request.headers().firstValue("Accept").isEmpty());
                        assertEquals("json_object", sent.at("/response_format/type").asText());
                        assertFalse(sent.has("temperature"));
                        if (provider.equals("BAILIAN")) {
                            assertTrue(sent.path("enable_thinking").asBoolean());
                            assertEquals(32_768, sent.path("thinking_budget").asInt());
                            assertFalse(sent.has("reasoning_effort"), "Bailian forbids sending both effort and thinking budget");
                            assertEquals(32_768, sent.path("max_tokens").asInt());
                        } else {
                            assertEquals("enabled", sent.at("/thinking/type").asText());
                            assertEquals("max", sent.path("reasoning_effort").asText());
                            assertEquals(65_536, sent.path("max_tokens").asInt());
                        }
                        return new BailianAiGateway.TransportResponse(200, envelope(input, "PRIVATE_REASONING_".repeat(5000)));
                    });
            var result = gateway.attempt(input);
            assertEquals(ReportGateway.Status.PROPOSAL_RECEIVED, result.status());
            assertTrue(result.telemetry().thinkingEnabled());
            assertEquals(5000L, result.telemetry().reasoningTokens());
            assertFalse(result.toString().contains("PRIVATE_REASONING"));
            assertEquals("report-model-input-v1", result.telemetry().inputSchemaVersion());
        }
    }

    @Test void boundedSubscriberAllowsLongReasoningButCancelsPastConfiguredCap() throws Exception {
        var requests = new AtomicInteger(); var cancellations = new AtomicInteger();
        Flow.Subscription subscription = new Flow.Subscription() {
            public void request(long count) { requests.incrementAndGet(); }
            public void cancel() { cancellations.incrementAndGet(); }
        };
        var allowed = new ReportHttpTransport.LimitedBody(1_048_576);
        allowed.onSubscribe(subscription); allowed.onNext(List.of(ByteBuffer.wrap(new byte[80_000]))); allowed.onComplete();
        assertEquals(80_000, allowed.getBody().toCompletableFuture().get().length);
        assertEquals(0, cancellations.get());
        var rejected = new ReportHttpTransport.LimitedBody(262_144);
        rejected.onSubscribe(subscription); rejected.onNext(List.of(ByteBuffer.wrap(new byte[262_145])));
        var failure = assertThrows(java.util.concurrent.ExecutionException.class, () -> rejected.getBody().toCompletableFuture().get());
        assertEquals("OUTPUT_LIMIT", failure.getCause().getMessage());
        assertEquals(1, cancellations.get());
    }

    @Test void strictSchemaBindsEachReportTypeForInitialAndRepairWithoutChangingModelOrContext() {

    @Test void v2OtherOutputModesAreExplicitAndSupplierRejectionNeverTriggersFallback() {

        var input = new ReportGateway.Attempt("format-no-fallback", ReportGateway.Stage.INITIAL, core.modelInput(catalog), List.of(), ReportModelSettings.MAX_TOTAL_BUDGET_MS);

    @Test void v2UnknownRootOrContextFieldsAndOversizeInputNeverReachProvider() {
        var catalog = core.compile(ReportAnalysisFixtures.v2(experience("BUSINESS_ANALYTICS", true), ReportAnalysisFixtures.SOUND));
        for (String kind : List.of("ROOT", "CONTEXT", "SIZE")) {
            var content = (ObjectNode) core.modelInput(catalog).content();
            var context = (ObjectNode) content.at("/sources/0/context");
            context.putArray("verifiedExecutions");
            if (kind.equals("ROOT")) content.put("password", "PRIVATE_SENTINEL");
            else if (kind.equals("CONTEXT")) context.put("credentials", "PRIVATE_SENTINEL");
            else ((ObjectNode) context.path("task")).put("title", "x".repeat(1_048_577));
            var input = new ReportGateway.Attempt("invalid-analysis", ReportGateway.Stage.INITIAL,
                    new ReportModels.SafeModelInput("report-model-input-v2", catalog.catalogSha256(), content), List.of(), ReportModelSettings.MAX_TOTAL_BUDGET_MS);
            var calls = new AtomicInteger();
            var gateway = new ReportHttpGateway(JSON, core, "BAILIAN", "synthetic-key", BailianAiGateway.DEFAULT_BASE_URL,
                    ReportHttpGateway.BAILIAN_MODEL, (request, timeout) -> { calls.incrementAndGet(); throw new AssertionError("Invalid input reached supplier"); });
            var result = gateway.attempt(input);
            assertEquals("INVALID_INPUT", result.safeErrorCode());
            assertEquals(0, calls.get());
            assertFalse(result.toString().contains("PRIVATE_SENTINEL"));
        }
